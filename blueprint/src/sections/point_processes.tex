\chapter{Vague convergence and Poisson random measures}
\label{chap:point-processes}

This chapter builds general infrastructure for the point-process approach to extremes, following
Resnick, \emph{Extreme Values, Regular Variation, and Point Processes}, Springer (1987),
Chapter~3 [Resnick], and Kallenberg, \emph{Random Measures, Theory and Applications}, Springer
(2017) [Kallenberg]. It lives in \texttt{ForMathlib/} and depends only on Mathlib, so that it can
be upstreamed.

\section{Vague convergence}

\begin{definition}[Vague convergence]
  \label{def:vague}
  \lean{MeasureTheory.Measure.TendstoVaguely}
  \leanok
  Measures $\mu_i$ on a topological space converge vaguely to $\mu$ if
  $\int f \, \ud\mu_i \to \int f \, \ud\mu$ for every continuous, compactly supported
  $f \colon X \to \bR$ [Resnick, \S3.4], [Kallenberg, \S4.1].
\end{definition}

\begin{theorem}[Basic properties of vague convergence]
  \label{thm:vague-portmanteau}
  \uses{def:vague}
  \lean{MeasureTheory.Measure.ext_of_forall_integral_eq_of_hasCompactSupport, MeasureTheory.Measure.TendstoVaguely.unique,
    MeasureTheory.Measure.TendstoVaguely.limsup_measure_le_of_isCompact,
    MeasureTheory.Measure.TendstoVaguely.le_liminf_measure_of_isOpen,
    MeasureTheory.Measure.TendstoVaguely.tendsto_measure_of_null_frontier,
    MeasureTheory.FiniteMeasure.tendstoVaguely_of_tendsto, MeasureTheory.ProbabilityMeasure.tendstoVaguely_of_tendsto}
  \leanok
  For regular measures on a locally compact Hausdorff space: vague limits are unique;
  $\limsup \mu_i(K) \le \mu(K)$ for compact $K$; $\mu(U) \le \liminf \mu_i(U)$ for open $U$; and
  $\mu_i(B) \to \mu(B)$ when $\overline B$ is compact and $\mu(\partial B) = 0$. Weak convergence of
  finite or probability measures implies vague convergence [Kallenberg, Lemma~4.1].
\end{theorem}
\begin{proof}
  \leanok
  Approximate indicators of compact and open sets from above and below by continuous compactly
  supported functions (Urysohn), and use regularity.
\end{proof}

\section{Poisson random measures}

\begin{definition}[Poisson random measure]
  \label{def:prm}
  \lean{ProbabilityTheory.IsPoissonRandomMeasure, ProbabilityTheory.IsPoissonRandomMeasure.hasLaw,
    ProbabilityTheory.IsPoissonRandomMeasure.iIndepFun}
  \leanok
  A measurable map $N \colon \Omega \to \mathcal M(E)$ is a Poisson random measure with intensity
  $\nu$ if for disjoint measurable $A_1, \dots, A_n$ of finite $\nu$-measure the counts
  $N(A_1), \dots, N(A_n)$ are independent and $N(A_i)$ is Poisson with mean $\nu(A_i)$
  [Resnick, \S3.3], [Kallenberg, \S3.2].
\end{definition}

\begin{theorem}[Existence of Poisson random measures]
  \label{thm:prm-exists}
  \uses{def:prm}
  \lean{ProbabilityTheory.sumDiracRange, ProbabilityTheory.poissonSampleMeasure, ProbabilityTheory.isPoissonRandomMeasure_sumDiracRange,
    ProbabilityTheory.isPoissonRandomMeasure_sum_sumDiracRange, ProbabilityTheory.exists_isPoissonRandomMeasure}
  \leanok
  For a finite intensity $c\mu$ with $\mu$ a probability measure, $N = \sum_{k < K} \delta_{Z_k}$ with
  $K \sim \mathrm{Poisson}(c)$ independent of iid $Z_k \sim \mu$ is a Poisson random measure. A
  superposition of independent such measures is a Poisson random measure with the summed intensity,
  so Poisson random measures exist for every s-finite (in particular every $\sigma$-finite)
  intensity [Kallenberg, Theorem~3.6].
\end{theorem}
\begin{proof}
  \leanok
  Conditionally on $K$, the counts are multinomial; the joint law is identified through
  characteristic functions. For superpositions, independent Poisson counts add.
\end{proof}

\begin{theorem}[Laplace functional]
  \label{thm:prm-laplace}
  \uses{thm:prm-exists}
  \lean{ProbabilityTheory.IsPoissonRandomMeasure.integral_exp_neg_sum, ProbabilityTheory.integral_exp_neg_integral_sumDiracRange}
  \leanok
  $\E \exp(-\int f \, \ud N) = \exp(-\int (1 - e^{-f}) \, \ud\nu)$ for simple $f \ge 0$ and every
  Poisson random measure, and for every measurable $f \ge 0$ for the finite-intensity
  construction [Resnick, Proposition~3.6].
\end{theorem}
\begin{proof}
  \leanok
  For simple $f$, independence and the Poisson Laplace transform. For the construction, condition on
  $K$: $\E e^{-\int f \ud N} = \E (\int e^{-f} \ud\mu)^K$.
\end{proof}

\begin{theorem}[Binomial point processes converge to Poisson]
  \label{thm:binomial-to-poisson}
  \uses{thm:prm-laplace, thm:vague-portmanteau}
  \lean{ProbabilityTheory.integral_exp_neg_integral_sumDiracRange_infinitePi,
    ProbabilityTheory.tendsto_laplace_sumDiracRange_of_tendstoVaguely,
    ProbabilityTheory.tendsto_laplace_sumDiracRange_poissonSampleMeasure}
  \leanok
  Let $N_n = \sum_{k<n} \delta_{Z_{n,k}}$ with $Z_{n,k}$ iid of law $\mu_n$. If $n \mu_n \to \nu$
  vaguely, then for continuous compactly supported $f \ge 0$,
  $\E \exp(-\int f \ud N_n) \to \exp(-\int (1 - e^{-f}) \ud\nu)$ [Resnick, Proposition~3.21].
  Convergence in law of the random measures, which needs a topology on $\mathcal M(E)$, is not yet
  formalized.
\end{theorem}
\begin{proof}
  \leanok
  $\E e^{-\int f \ud N_n} = (1 - \int (1 - e^{-f}) \ud\mu_n)^n$, and $n \int (1 - e^{-f}) \ud\mu_n \to
  \int (1 - e^{-f}) \ud\nu$ by vague convergence.
\end{proof}

\section{Poisson limits, uniqueness and convergence of point processes}

\begin{theorem}[Multivariate Poisson limit for triangular arrays]
  \label{thm:poisson-limit}
  \lean{ProbabilityTheory.integral_torusMeasure_charFun_mul, ProbabilityTheory.tendsto_measureReal_eq_of_tendsto_integral_cexp,
    ProbabilityTheory.tendsto_prod_range_one_add, ProbabilityTheory.tendsto_measureReal_card_filter_range_eq}
  \leanok
  Let $Z_{n,i}$, $i < n$, be independent for each $n$, and $B_1, \dots, B_k$ disjoint with
  $\sum_{i<n} \PR[Z_{n,i} \in B_j] \to r_j$ and $\max_{i<n} \PR[Z_{n,i} \in B_j] \to 0$. Then the counts
  $(\#\{i : Z_{n,i} \in B_j\})_j$ converge in law to independent Poisson$(r_j)$ variables.
\end{theorem}
\begin{proof}
  \leanok
  Characteristic functions of the count vectors converge; point probabilities of $\bN^k$-valued
  vectors are recovered by Fourier inversion over the torus $(0, 2\pi]^k$.
\end{proof}

\begin{theorem}[Laplace functional and uniqueness]
  \label{thm:prm-uniqueness}
  \uses{thm:prm-laplace}
  \lean{ProbabilityTheory.IsPoissonRandomMeasure.lintegral_exp_neg_lintegral,
    ProbabilityTheory.IsPoissonRandomMeasure.lintegral_exp_neg_lintegral_simpleFunc,
    ProbabilityTheory.ext_of_forall_integral_prod_pow_eq, ProbabilityTheory.map_eq_of_lintegral_exp_neg_sum_eq,
    ProbabilityTheory.map_eq_of_lintegral_exp_neg_lintegral_eq, ProbabilityTheory.IsPoissonRandomMeasure.map_eq}
  \leanok
  For $\sigma$-finite $\nu$, every Poisson random measure satisfies
  $\E e^{-\int f \ud N} = e^{-\int (1 - e^{-f}) \ud\nu}$ for all measurable $f \ge 0$. The law of a
  random measure is determined by its Laplace functional on simple functions, so all Poisson random
  measures with the same intensity have the same law [Kallenberg, Lemma~3.1].
\end{theorem}
\begin{proof}
  \leanok
  Monotone approximation by simple functions. For uniqueness, $E \exp(-\sum a_i N(A_i))$ for
  $a \in \bN^k$ are moments of $(e^{-N(A_i)})_i \in [0,1]^k$, and finite measures on the cube are
  determined by their moments.
\end{proof}

\begin{definition}[Convergence in law of point processes]
  \label{def:tendsto-in-law-vaguely}
  \uses{def:vague}
  \lean{ProbabilityTheory.TendstoInLawVaguely, ProbabilityTheory.IsPoissonRandomMeasure.measureReal_forall_apply_eq}
  \leanok
  $N_n \Rightarrow N$ if for every finite family of disjoint relatively compact $\nu$-continuity sets
  $B_j$, the joint laws of the counts $(N_n(B_j))_j$ converge to those of $(N(B_j))_j$
  [Kallenberg, Theorem~4.11].
\end{definition}

\begin{theorem}[Binomial point processes converge to Poisson]
  \label{thm:binomial-to-prm}
  \uses{thm:poisson-limit, def:tendsto-in-law-vaguely, thm:binomial-to-poisson}
  \lean{ProbabilityTheory.tendstoInLawVaguely_sumDiracRange}
  \leanok
  If $n\mu_n \to \nu$ vaguely, the binomial point processes $\sum_{i<n} \delta_{Z_{n,i}}$,
  $Z_{n,i} \sim \mu_n$ iid, converge in law to the Poisson random measure with intensity $\nu$
  [Resnick, Proposition~3.21].
\end{theorem}
\begin{proof}
  \leanok
  Theorem~\ref{thm:poisson-limit} with $r_j = \nu(B_j)$, using the portmanteau property of vague
  convergence on $\nu$-continuity sets.
\end{proof}

\section{The point process of exceedances}

\begin{theorem}[Exceedances converge to a Poisson process]
  \label{thm:exceedances}
  \uses{thm:poisson-limit, thm:kth-largest, thm:two-largest}
  \lean{ExtremeValueTheory.tendsto_nat_mul_measureReal_lt_affine, ExtremeValueTheory.tendsto_nat_mul_measureReal_affine_mem_Ioc,
    ExtremeValueTheory.tendsto_measureReal_card_affine_mem_eq, ExtremeValueTheory.exceedanceMeasure, ExtremeValueTheory.IsExceedanceInterval,
    ExtremeValueTheory.tendsto_measureReal_sumDiracRange_affine_eq, ExtremeValueTheory.tendsto_measureReal_sumDiracRange_timeSpace_eq,
    ExtremeValueTheory.tendsto_measureReal_affine_orderStat_le_of_pointProcess,
    ExtremeValueTheory.tendsto_measureReal_affine_orderStat_zero_one_of_pointProcess}
  \leanok
  Let $X_i$ be iid with $F^n(a_n x + b_n) \to G(x)$. The point processes
  $\sum_{i<n} \delta_{(X_i - b_n)/a_n}$ and $\sum_{i<n} \delta_{((i+1)/n, (X_i - b_n)/a_n)}$ converge, on
  disjoint exceedance intervals above a threshold $u$ with $G(u) > 0$ (resp.\ rectangles in time),
  to Poisson random measures with intensity $\mu_G(a, b] = \log G(b) - \log G(a)$
  (resp.\ $\mathrm{Leb} \otimes \mu_G$) [Resnick, Corollary~4.19], [EKM, Theorem~5.3.2]. As
  corollaries, the limit laws of the $k$-th largest and of the two largest are recovered,
  matching Theorems~\ref{thm:kth-largest} and~\ref{thm:two-largest}.
\end{theorem}
\begin{proof}
  \leanok
  $n \PR[(X - b_n)/a_n > x] \to -\log G(x)$, then Theorem~\ref{thm:poisson-limit}.
\end{proof}
