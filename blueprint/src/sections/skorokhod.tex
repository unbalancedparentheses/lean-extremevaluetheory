\chapter{The Skorokhod space}
\label{chap:skorokhod}

General infrastructure for functional limit theorems, following Billingsley, \emph{Convergence of
Probability Measures}, 2nd ed., Wiley (1999), \S\S12--13 [Billingsley]. It lives in \texttt{ForMathlib/}
for upstreaming.

\begin{lemma}[Structure of c\`adl\`ag functions]
  \label{lem:cadlag}
  \lean{IsCadlag.exists_finset_edist_lt, IsCadlag.exists_monotone_finite_range_edist_lt,
    IsCadlag.exists_finite_range_edist_lt, IsCadlag.finite_setOf_lt_edist_leftLim,
    IsCadlag.countable_setOf_edist_leftLim_ne_zero,
    IsCadlag.countable_setOf_leftLim_ne, IsCadlag.measurable}
  \leanok
  A c\`adl\`ag function on a compact interval admits, for every $\varepsilon > 0$, a finite partition on whose cells it
  oscillates less than $\varepsilon$ [Billingsley, Lemma~12.1]; hence it is a uniform limit of step functions, has
  finitely many jumps larger than $\varepsilon$ and at most countably many jumps, and is Borel measurable.
\end{lemma}
\begin{proof}
  \leanok
  Compactness and the existence of one-sided limits.
\end{proof}

\begin{definition}[The Skorokhod $J_1$ metric]
  \label{def:skorokhod}
  \uses{lem:cadlag}
  \lean{skorokhodEDistWith, skorokhodEDist, SkorokhodSpace}
  \leanok
  For $x, y \colon [0,1] \to E$, $d(x, y) = \inf_\lambda \max(\|\lambda - \mathrm{id}\|_\infty, \sup_t d_E(x(t), y(\lambda t)))$ over
  increasing homeomorphisms $\lambda$ of $[0,1]$. The Skorokhod space $D([0,1], E)$ is the space of c\`adl\`ag functions
  with this distance [Billingsley, \S12].
\end{definition}

\begin{theorem}[$J_1$ is a metric]
  \label{thm:skorokhod-metric}
  \uses{def:skorokhod}
  \lean{skorokhodEDist_self, skorokhodEDist_comm, skorokhodEDist_triangle, skorokhodEDist_le_iSup_edist,
    edist_eq_zero_of_skorokhodEDist_eq_zero, SkorokhodSpace.tendsto_iff_tendstoUniformly,
    SkorokhodSpace.tendsto_of_tendstoUniformly, SkorokhodSpace.tendsto_apply_of_continuousAt,
    SkorokhodSpace.lipschitzWith_ofContinuousMap}
  \leanok
  $d$ is a metric on $D$, bounded by the uniform distance. $J_1$-convergence to a continuous limit is uniform
  convergence, $J_1$-convergence gives convergence at continuity points of the limit, and $C([0,1], E)$ embeds
  $1$-Lipschitz into $D$ [Billingsley, \S12].
\end{theorem}
\begin{proof}
  \leanok
  Compose and invert time changes; distance $0$ forces equality by right-continuity.
\end{proof}

\begin{theorem}[$D$ is separable and its Borel $\sigma$-algebra is generated by the projections]
  \label{thm:skorokhod-borel}
  \uses{thm:skorokhod-metric}
  \lean{SkorokhodSpace.instSeparableSpace, SkorokhodSpace.measurable_apply, SkorokhodSpace.measurable_coeFn,
    SkorokhodSpace.measurableSpace_eq_comap_coeFn, SkorokhodSpace.measurableSpace_eq_iSup_comap}
  \leanok
  For separable $E$, $D([0,1], E)$ is separable [Billingsley, Theorem~12.2]. Every projection $\pi_t$ is Borel, and for
  second-countable $E$ the Borel $\sigma$-algebra of $D$ equals the $\sigma$-algebra generated by the projections
  [Billingsley, Theorem~12.5].
\end{theorem}
\begin{proof}
  \leanok
  Step functions with rational jump times and values in a countable dense set are dense, using time changes
  that move finitely many points to rationals. Projections are measurable via upper semicontinuity of
  $x \mapsto \inf_{s \in (a,b)} g(x(s))$; conversely paths are approximated by grid step functions.
\end{proof}

\begin{theorem}[$D$ is Polish]
  \label{thm:skorokhod-polish}
  \uses{thm:skorokhod-borel}
  \lean{unitInterval.logSlopeNorm, billingsleyEDist, billingsleyEDist_le_iSup_edist,
    skorokhodEDist_le_exp_billingsleyEDist_sub_one, IsCadlag.exists_billingsleyEDist_lt,
    SkorokhodSpace.homeomorphBillingsley, SkorokhodSpace.Billingsley.instCompleteSpace,
    SkorokhodSpace.instIsCompletelyMetrizableSpace, SkorokhodSpace.instPolishSpace,
    IsCadlag.of_tendstoUniformly, IsCadlag.comp_orderIso}
  \leanok
  Billingsley's metric $d^\circ(x, y) = \inf_\lambda \max(\|\lambda\|^\circ, \|x - y \circ \lambda\|_\infty)$, with the log-slope norm
  $\|\lambda\|^\circ = \sup_{s < t} |\log((\lambda t - \lambda s)/(t - s))|$, is a metric on $D$ inducing the $J_1$ topology, and $(D, d^\circ)$ is
  complete for complete $E$ [Billingsley, Theorems~12.1 and~12.2]. Hence $D([0,1], E)$ is Polish for every complete
  separable metric space $E$.
\end{theorem}
\begin{proof}
  \leanok
  $d \le e^{d^\circ} - 1$, and conversely $d$-small implies $d^\circ$-small at a fixed c\`adl\`ag function. Completeness: a $d^\circ$-Cauchy
  sequence has a subsequence with summable distances; compose the time changes and pass to a uniform limit, which
  is c\`adl\`ag.
\end{proof}

\section{\texorpdfstring{Weak convergence in $D$ and the tail empirical process}{Weak convergence in D and the tail empirical process}}

\begin{theorem}[Weak convergence in $D$]
  \label{thm:skorokhod-weak-convergence}
  \uses{thm:skorokhod-polish}
  \lean{SkorokhodSpace.tendsto_of_tendsto_map_restrict_of_isTight,
    SkorokhodSpace.exists_tendsto_of_tendsto_map_restrict_of_isTight,
    SkorokhodSpace.isTightMeasureSet_of_cadlagModulus, SkorokhodSpace.isTightMeasureSet_of_blockOscillation}
  \leanok
  Tight laws on $D$ whose finite-dimensional distributions converge outside a countable set of times (with $1$ included)
  converge weakly [Billingsley, Theorem~13.1]; tightness follows from a boundedness condition and the modulus condition
  $w'_x(\delta) \to 0$ [Billingsley, Theorem~13.2].
\end{theorem}
\begin{proof}
  \leanok
  Compactness in $D$ via the modulus $w'$ (Billingsley, Theorem~12.3), Prokhorov, and determination of laws on $D$ by the
  finite-dimensional distributions on a dense set of times.
\end{proof}

\begin{theorem}[The uniform tail empirical process converges to Brownian motion]
  \label{thm:tail-empirical-process}
  \uses{thm:skorokhod-weak-convergence}
  \lean{ExtremeValueTheory.tendstoInDistribution_tailEmpiricalProcess_fdd,
    ExtremeValueTheory.exists_tendsto_tailEmpiricalPath, ExtremeValueTheory.tendstoInDistribution_tailEmpiricalPath,
    ExtremeValueTheory.tendsto_tailEmpiricalPath_of_fdd}
  \leanok
  For iid uniform $U_i$, $k \to \infty$ and $k/n \to 0$, the process
  $W_n(t) = \sqrt k\,\big(k^{-1}\sum_{i<n} \mathbf 1\{U_i \ge 1 - (k/n)t\} - t\big)$, $t \in [0,1]$, converges weakly in $D[0,1]$ to Brownian
  motion [dHF, Theorem~5.1.4], [Einmahl 1992]. In particular a Brownian law on $D$ exists. The weighted version and the
  corresponding process for regularly varying $F$ (which gives a second proof of Hill normality) are not formalized.
\end{theorem}
\begin{proof}
  \leanok
  Finite-dimensional convergence by a triangular-array CLT. Tightness: $\E(W_n(t) - W_n(s))^4 \le 12(t-s)^2 + 2(t-s)/k$,
  dyadic chaining, and monotonicity of the counts between grid points.
\end{proof}
