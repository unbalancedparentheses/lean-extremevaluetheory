\chapter{The Skorokhod space}
\label{chap:skorokhod}

General infrastructure for functional limit theorems, following Billingsley, \emph{Convergence of
Probability Measures}, 2nd ed., Wiley (1999), \S\S12--13 [Billingsley]. It lives in \texttt{ForMathlib/}
for upstreaming.

\begin{lemma}[Structure of c\`adl\`ag functions]
  \label{lem:cadlag}
  \lean{IsCadlag.exists_finset_edist_lt, IsCadlag.exists_monotone_finite_range_edist_lt,
    IsCadlag.exists_finite_range_edist_lt, IsCadlag.finite_setOf_lt_edist_leftLim,
    IsCadlag.countable_setOf_edist_leftLim_ne_zero,
    IsCadlag.countable_setOf_leftLim_ne, IsCadlag.measurable}
  \leanok
  A c\`adl\`ag function on a compact interval admits, for every $\varepsilon > 0$, a finite partition on whose cells it
  oscillates less than $\varepsilon$ [Billingsley, Lemma~12.1]; hence it is a uniform limit of step functions, has
  finitely many jumps larger than $\varepsilon$ and at most countably many jumps, and is Borel measurable.
\end{lemma}
\begin{proof}
  \leanok
  Compactness and the existence of one-sided limits.
\end{proof}

\begin{definition}[The Skorokhod $J_1$ metric]
  \label{def:skorokhod}
  \uses{lem:cadlag}
  \lean{skorokhodEDistWith, skorokhodEDist, SkorokhodSpace}
  \leanok
  For $x, y \colon [0,1] \to E$, $d(x, y) = \inf_\lambda \max(\|\lambda - \mathrm{id}\|_\infty, \sup_t d_E(x(t), y(\lambda t)))$ over
  increasing homeomorphisms $\lambda$ of $[0,1]$. The Skorokhod space $D([0,1], E)$ is the space of c\`adl\`ag functions
  with this distance [Billingsley, \S12].
\end{definition}

\begin{theorem}[$J_1$ is a metric]
  \label{thm:skorokhod-metric}
  \uses{def:skorokhod}
  \lean{skorokhodEDist_self, skorokhodEDist_comm, skorokhodEDist_triangle, skorokhodEDist_le_iSup_edist,
    edist_eq_zero_of_skorokhodEDist_eq_zero, SkorokhodSpace.tendsto_iff_tendstoUniformly,
    SkorokhodSpace.tendsto_of_tendstoUniformly, SkorokhodSpace.tendsto_apply_of_continuousAt,
    SkorokhodSpace.lipschitzWith_ofContinuousMap}
  \leanok
  $d$ is a metric on $D$, bounded by the uniform distance. $J_1$-convergence to a continuous limit is uniform
  convergence, $J_1$-convergence gives convergence at continuity points of the limit, and $C([0,1], E)$ embeds
  $1$-Lipschitz into $D$ [Billingsley, \S12].
\end{theorem}
\begin{proof}
  \leanok
  Compose and invert time changes; distance $0$ forces equality by right-continuity.
\end{proof}

\begin{theorem}[$D$ is separable and its Borel $\sigma$-algebra is generated by the projections]
  \label{thm:skorokhod-borel}
  \uses{thm:skorokhod-metric}
  \lean{SkorokhodSpace.instSeparableSpace, SkorokhodSpace.measurable_apply, SkorokhodSpace.measurable_coeFn,
    SkorokhodSpace.measurableSpace_eq_comap_coeFn, SkorokhodSpace.measurableSpace_eq_iSup_comap}
  \leanok
  For separable $E$, $D([0,1], E)$ is separable [Billingsley, Theorem~12.2]. Every projection $\pi_t$ is Borel, and for
  second-countable $E$ the Borel $\sigma$-algebra of $D$ equals the $\sigma$-algebra generated by the projections
  [Billingsley, Theorem~12.5]. Completeness under Billingsley's metric $d^\circ$ (so that $D$ is Polish) is not
  formalized.
\end{theorem}
\begin{proof}
  \leanok
  Step functions with rational jump times and values in a countable dense set are dense, using time changes
  that move finitely many points to rationals. Projections are measurable via upper semicontinuity of
  $x \mapsto \inf_{s \in (a,b)} g(x(s))$; conversely paths are approximated by grid step functions.
\end{proof}
