\chapter{Domains of attraction}
\label{chap:domains-of-attraction}

A c.d.f.\ $F$ is in the \emph{maximum domain of attraction} of an extreme value
distribution $G$ if there are orientation preserving affine isomorphisms $A_n \in \OriAffR$
with $\big( (A_n . F)(x) \big)^n \to G(x)$ for every $x$, i.e., if the suitably normalized
maximum of $n$ i.i.d.\ samples from $F$ converges in distribution to $G$. This chapter proves
Gnedenko's characterizations of the Fréchet and Weibull domains of attraction in terms of
regular variation, and de Haan's characterization of the Gumbel domain of attraction in terms
of an auxiliary function. References: de Haan and Ferreira, \emph{Extreme Value Theory: An
Introduction}, Springer (2006) [dHF]; Resnick, \emph{Extreme Values, Regular Variation, and
Point Processes}, Springer (1987) [Resnick]; Embrechts, Klüppelberg and Mikosch,
\emph{Modelling Extremal Events}, Springer (1997) [EKM].

\section{The Fréchet domain of attraction}

\begin{definition}[Fréchet maximum domain of attraction]
  \label{def:mda-frechet}
  \uses{def:cdf, def:std-Frechet-cdf, lem:oriented-affine-action-on-cdf}
  \lean{CumulativeDistributionFunction.IsInMDAFrechet}
  \leanok
  Let $\alpha > 0$. A c.d.f.\ $F$ is in the maximum domain of attraction of the standard
  Fréchet distribution $\FrechetCDF{\alpha}$ (with $\FrechetCDF{\alpha}(x) = \exp(-x^{-\alpha})$
  for $x > 0$ and $0$ for $x \le 0$) if there are $A_n \in \OriAffR$ with
  $\big( (A_n . F)(x) \big)^n \to \FrechetCDF{\alpha}(x)$ for every $x \in \bR$.
  (Since $\FrechetCDF{\alpha}$ is continuous, this is the same as convergence at all
  continuity points.)
\end{definition}

\begin{lemma}[Scaling constants from regular variation]
  \label{lem:frechet-exists-scaling}
  \uses{def:regularly-varying}
  \lean{ExtremeValueTheory.Frechet.exists_tendsto_mul_of_isRegularlyVarying}
  \leanok
  Let $g$ be antitone, right-continuous, with $g(x) \to 0$ as $x \to \infty$, and regularly
  varying with index $-\alpha$. Then there are constants $a_n > 0$ such that
  $n \, g(a_n x) \to x^{-\alpha}$ for every $x > 0$.
\end{lemma}
\begin{proof}
  \leanok
  Take the quantile-type constants $a_n = \inf \set{ x \; | \; g(x) \le 1/n }$ (made positive).
  Right continuity gives $n \, g(a_n) \le 1$, and regular variation at $c \, a_n$ with $c < 1$
  gives $\liminf n \, g(a_n) \ge c^{-\alpha}$ for every $c<1$; hence $n \, g(a_n) \to 1$ (using
  $a_n \to \infty$, which holds as $g > 0$ eventually). Then
  $n \, g(a_n x) = n \, g(a_n) \cdot \frac{g(x a_n)}{g(a_n)} \to x^{-\alpha}$.
\end{proof}

\begin{lemma}[Regular variation from affine normalization]
  \label{lem:frechet-rv-of-affine}
  \uses{def:regularly-varying}
  \lean{ExtremeValueTheory.Frechet.isRegularlyVarying_of_tendsto_mul_affine}
  \leanok
  Let $g$ be antitone, $\alpha > 0$, and $a_n > 0$, $b_n \in \bR$ with
  $n \, g(a_n y + b_n) \to y^{-\alpha}$ for every $y > 0$. Then $g$ is regularly varying with
  index $-\alpha$.
\end{lemma}
\begin{proof}
  \leanok
  Extend $a, b$ to real arguments by $a(t) = a_{\lfloor t \rfloor}$, $b(t) = b_{\lfloor t \rfloor}$,
  so that $t \, g(a(t) y + b(t)) \to y^{-\alpha}$. Comparing the levels $t$ and $2t$ by
  monotonicity of $g$ shows that $a(2t) \ge \rho \, a(t)$ eventually for some $\rho > 1$ (indeed
  $a(2t) \sim 2^{1/\alpha} a(t)$) and that $b(2t) - b(t) = o(a(t))$. A doubling argument
  (summing along dyadic chains $t, 2t, 4t, \ldots$ and using local bounds on a base interval
  $[T, 2T]$) then yields $a(t) \to \infty$ and $b(t)/a(t) \to 0$. Hence the centring may be
  dropped, $n \, g(a_n x) \to x^{-\alpha}$ for $x > 0$. Finally, for $t$ large choose
  $N(t) = \sup \set{ n \; | \; a_n \le t }$, so that $a_{N(t)} \le t < a_{N(t)+1}$; a sandwich
  using monotonicity of $g$ gives $g(c t)/g(t) \to c^{-\alpha}$.
\end{proof}

\begin{theorem}[Gnedenko's theorem, Fréchet case]
  \label{thm:mda-frechet-iff}
  \uses{def:mda-frechet, def:regularly-varying, lem:frechet-exists-scaling,
    lem:frechet-rv-of-affine, thm:tfae-ev-limit}
  \lean{CumulativeDistributionFunction.isInMDAFrechet_iff,
    CumulativeDistributionFunction.isInMDAFrechet_of_isRegularlyVarying,
    CumulativeDistributionFunction.isRegularlyVarying_of_isInMDAFrechet}
  \leanok
  Let $\alpha > 0$. A c.d.f.\ $F$ is in the maximum domain of attraction of $\FrechetCDF{\alpha}$
  if and only if its tail $x \mapsto 1 - F(x)$ is regularly varying with index $-\alpha$
  [dHF, Theorem~1.2.1], [Resnick, Proposition~1.11], [EKM, Theorem~3.3.7].
\end{theorem}
\begin{proof}
  \leanok
  Sufficiency: the tail $g = 1 - F$ is antitone, right-continuous and tends to $0$, so
  Lemma~\ref{lem:frechet-exists-scaling} gives $a_n > 0$ with $n (1 - F(a_n x)) \to x^{-\alpha}
  = -\log \FrechetCDF{\alpha}(x)$ for $x > 0$. With $A_n^{-1}(x) = a_n x$, the equivalence of
  the limit relations (Theorem~\ref{thm:tfae-ev-limit}) gives
  $(A_n . F)(x)^n \to \FrechetCDF{\alpha}(x)$ for $x > 0$. For $x \le 0$ we use
  $F(a_n x) \le F(0) < 1$ (as $1 - F > 0$ everywhere), so $F(a_n x)^n \to 0$.

  Necessity: write $A_n^{-1}(x) = a_n x + b_n$ with $a_n > 0$. By
  Theorem~\ref{thm:tfae-ev-limit}, $n (1 - F(a_n y + b_n)) \to y^{-\alpha}$ for $y > 0$, and
  Lemma~\ref{lem:frechet-rv-of-affine} gives regular variation of $1 - F$.
\end{proof}


\section{The Weibull domain of attraction}

\begin{definition}[Weibull maximum domain of attraction]
  \label{def:mda-weibull}
  \uses{def:cdf, def:std-Weibull-cdf, lem:oriented-affine-action-on-cdf}
  \lean{CumulativeDistributionFunction.IsInMDAWeibull}
  \leanok
  Let $\alpha > 0$. A c.d.f.\ $F$ is in the maximum domain of attraction of the standard
  Weibull distribution $\WeibullCDF{\alpha}$ (with $\WeibullCDF{\alpha}(x) = \exp(-(-x)^\alpha)$
  for $x < 0$ and $1$ for $x \ge 0$) if there are $A_n \in \OriAffR$ with
  $\big( (A_n . F)(x) \big)^n \to \WeibullCDF{\alpha}(x)$ for every $x \in \bR$.
\end{definition}

\begin{lemma}[Regularly varying antitone functions vanish at infinity]
  \label{lem:weibull-tendsto-zero}
  \uses{def:regularly-varying}
  \lean{ExtremeValueTheory.Weibull.tendsto_zero_of_isRegularlyVarying}
  \leanok
  If $g$ is antitone, positive, and regularly varying with index $-\alpha$, $\alpha > 0$,
  then $g(x) \to 0$ as $x \to \infty$.
\end{lemma}
\begin{proof}
  \leanok
  $g$ has a limit $\ell \ge 0$ at $+\infty$ by monotonicity. If $\ell > 0$, then
  $g(2x)/g(x) \to \ell/\ell = 1 \ne 2^{-\alpha}$, a contradiction.
\end{proof}

\begin{lemma}[The endpoint lemma]
  \label{lem:weibull-endpoint}
  \lean{ExtremeValueTheory.Weibull.eventually_eq_zero_of_tendsto_mul_affine}
  \leanok
  Let $u$ be antitone and nonnegative, $\alpha > 0$, $a_n > 0$, $b_n \in \bR$, with
  $n \, u(a_n z + b_n) \to (-z)^\alpha$ for every $z < 0$ and $n \, u(b_n) \to 0$. Then for
  every $c > 0$, eventually $u(b_n + c \, a_n) = 0$.
\end{lemma}
\begin{proof}
  \leanok
  Let $\kappa = 2^{-1/\alpha} < 1$. Comparing the levels $n$ and $2n$ by monotonicity of $u$
  (using $2n \, u(a_{2n} z + b_{2n}) \to 2(-z)^\alpha$) shows that, for every small $\delta > 0$,
  eventually $a_{2n} \le \rho \, a_n$ with $\rho = \kappa + 2\delta < 1$ and
  $b_{2n} - b_n \le 3 \delta \, a_n$. Choosing $\delta$ small relative to $c (1 - \rho)$, the
  potential $\Phi_n = b_n + c \, a_n$ is eventually non-increasing along doublings,
  $\Phi_{2n} \le \Phi_n$. Suppose $\eta = u(\Phi_n) > 0$ for some large $n$. For $m = 2^k n$ we get
  $b_m < \Phi_m \le \Phi_n$, so $u(b_m) \ge \eta$ and $m \, u(b_m) \ge \eta$ for all $k$,
  contradicting $m \, u(b_m) \to 0$. Hence $u(\Phi_n) = 0$ eventually.
\end{proof}

\begin{theorem}[Gnedenko's theorem, Weibull case]
  \label{thm:mda-weibull-iff}
  \uses{def:mda-weibull, def:regularly-varying, lem:frechet-exists-scaling,
    lem:frechet-rv-of-affine, lem:weibull-tendsto-zero, lem:weibull-endpoint, thm:tfae-ev-limit}
  \lean{CumulativeDistributionFunction.isInMDAWeibull_iff,
    CumulativeDistributionFunction.isInMDAWeibull_of_isRegularlyVarying,
    CumulativeDistributionFunction.exists_endpoint_of_isInMDAWeibull}
  \leanok
  Let $\alpha > 0$. A c.d.f.\ $F$ is in the maximum domain of attraction of $\WeibullCDF{\alpha}$
  if and only if $F$ has a finite right endpoint $x_F$ (that is, $F(x_F) = 1$ and $F(x) < 1$ for
  $x < x_F$) and $t \mapsto 1 - F(x_F - t^{-1})$ is regularly varying with index $-\alpha$
  [dHF, Theorem~1.2.1(2)], [EKM, Theorem~3.3.12].
\end{theorem}
\begin{proof}
  \leanok
  The transformation $x \mapsto x_F - x^{-1}$ reduces both directions to the analytic lemmas
  of the Fréchet case, applied to the antitone function $g(t) = 1 - F(x_F - t^{-1})$
  (suitably extended to small $t$).

  Sufficiency: $g$ tends to $0$ by Lemma~\ref{lem:weibull-tendsto-zero}, and
  Lemma~\ref{lem:frechet-exists-scaling} gives $a_n$ with $n \, g(a_n y) \to y^{-\alpha}$ for
  $y > 0$. Taking $A_n^{-1}(x) = x_F + a_n^{-1} x$ and substituting $y = -x^{-1}$ for $x < 0$
  gives $n (1 - F(A_n^{-1}(x))) \to (-x)^\alpha$, hence the limit relation by
  Theorem~\ref{thm:tfae-ev-limit}; for $x \ge 0$, $F(A_n^{-1}(x)) \ge F(x_F) = 1$.

  Necessity: write $A_n^{-1}(x) = a_n x + b_n$. Theorem~\ref{thm:tfae-ev-limit} gives
  $n (1 - F(a_n z + b_n)) \to (-z)^\alpha$ for $z < 0$, while $F(b_n)^n \to 1$ gives
  $n (1 - F(b_n)) \to 0$. The endpoint lemma (Lemma~\ref{lem:weibull-endpoint}) shows that
  $F(b_n + c \, a_n) = 1$ eventually for every $c > 0$; in particular the right endpoint
  $x_F$ is finite, and one deduces $(b_n - x_F)/a_n \to 0$. The centring $b_n$ can then be
  replaced by $x_F$, and Lemma~\ref{lem:frechet-rv-of-affine}, applied after the
  transformation $x \mapsto x_F - x^{-1}$, gives regular variation of $g$.
\end{proof}


\section{The Gumbel domain of attraction}

Throughout this section $x_F = \sup \set{ x \; | \; F(x) < 1 } \in (-\infty, +\infty]$ denotes
the right endpoint of $F$. We say that \emph{$t \uparrow x_F$} for $t \to +\infty$ when
$x_F = +\infty$ (that is, $F < 1$ everywhere), and for $t \to x_F$ from the left when
$x_F < \infty$ (that is, $F(x) < 1$ for $x < x_F$ and $F(x_F) = 1$). An \emph{auxiliary function}
for $F$ is a function $f$, positive for $t$ near $x_F$, such that
\begin{align*}
  \lim_{t \uparrow x_F} \frac{1 - F(t + x f(t))}{1 - F(t)} \; = \; e^{-x}
  \qquad \text{for every } x \in \bR .
\end{align*}

\begin{definition}[Gumbel maximum domain of attraction]
  \label{def:mda-gumbel}
  \uses{def:cdf, def:std-Gumbel-cdf, lem:oriented-affine-action-on-cdf}
  \lean{CumulativeDistributionFunction.IsInMDAGumbel}
  \leanok
  A c.d.f.\ $F$ is in the maximum domain of attraction of the standard Gumbel distribution
  $\GumbelCDF(x) = \exp(-e^{-x})$ if there are $A_n \in \OriAffR$ with
  $\big( (A_n . F)(x) \big)^n \to \GumbelCDF(x)$ for every $x \in \bR$.
\end{definition}

\begin{lemma}[Gumbel membership from the tail limit relation]
  \label{lem:mda-gumbel-of-tendsto-mul}
  \uses{def:mda-gumbel, thm:tfae-ev-limit}
  \lean{CumulativeDistributionFunction.isInMDAGumbel_of_tendsto_mul,
    CumulativeDistributionFunction.tendsto_mul_of_isInMDAGumbel}
  \leanok
  $F$ is in the Gumbel domain of attraction if and only if there are $a_n > 0$ and $b_n \in \bR$
  with $n \big( 1 - F(a_n x + b_n) \big) \to e^{-x}$ for every $x \in \bR$.
\end{lemma}
\begin{proof}
  \leanok
  Since $-\log \GumbelCDF(x) = e^{-x}$ and $\GumbelCDF(x) \in (0, 1)$, this is the equivalence of
  the limit relations (Theorem~\ref{thm:tfae-ev-limit}) with $A_n^{-1}(x) = a_n x + b_n$.
\end{proof}

\begin{lemma}[Generalized quantiles]
  \label{lem:gumbel-quantile}
  \lean{ExtremeValueTheory.Gumbel.quantile_spec}
  \leanok
  Let $g$ be antitone and right-continuous, and let $c$ be such that $g(x) \le c$ for some $x$
  and $c < g(y)$ for some $y$. Then $q = \inf \set{ x \; | \; g(x) \le c }$ satisfies
  $g(q) \le c$ and $c < g(y)$ for every $y < q$.
\end{lemma}
\begin{proof}
  \leanok
  The set $\set{ x \; | \; g(x) \le c }$ is a nonempty up-set bounded below; right continuity
  gives $g(q) \le c$, and $y < q$ means $y$ is not in the set.
\end{proof}

\begin{lemma}[Sufficiency, analytic core]
  \label{lem:gumbel-sufficiency-core}
  \lean{ExtremeValueTheory.Gumbel.tendsto_mul_of_auxiliary}
  \leanok
  Let $L$ be a filter on $\bR$, and let $g, f$ be eventually positive along $L$ with
  $g(t + x f(t))/g(t) \to e^{-x}$ along $L$ for every $x$. Let $b_n \to L$ satisfy, eventually,
  $g(b_n) \le 1/n < g(y)$ for all $y < b_n$. Then
  $n \, g\big( b_n + x f(b_n) \big) \to e^{-x}$ for every $x$.
\end{lemma}
\begin{proof}
  \leanok
  First, $n \, g(b_n) \to 1$: the upper bound $n \, g(b_n) \le 1$ holds by assumption, while for
  $\delta > 0$ the point $b_n - \delta f(b_n) < b_n$ gives
  $1 < n \, g(b_n - \delta f(b_n)) = n \, g(b_n) \cdot g(b_n - \delta f(b_n))/g(b_n)$, and the
  ratio tends to $e^{\delta}$, so $\liminf n \, g(b_n) \ge e^{-\delta}$. Then
  $n \, g(b_n + x f(b_n)) = n \, g(b_n) \cdot g(b_n + x f(b_n))/g(b_n) \to e^{-x}$.
\end{proof}

\begin{theorem}[Sufficiency via an auxiliary function]
  \label{thm:mda-gumbel-of-auxiliary}
  \uses{def:mda-gumbel, lem:mda-gumbel-of-tendsto-mul, lem:gumbel-quantile,
    lem:gumbel-sufficiency-core}
  \lean{CumulativeDistributionFunction.tendsto_mul_one_sub_of_auxiliary_atTop,
    CumulativeDistributionFunction.isInMDAGumbel_of_auxiliary_atTop,
    CumulativeDistributionFunction.isInMDAGumbel_of_auxiliary_nhdsLT}
  \leanok
  If $F$ admits an auxiliary function $f$ (with either $x_F = +\infty$ or $x_F < \infty$),
  then $F$ is in the Gumbel domain of attraction, with normalizing constants
  $b_n = F^{\leftarrow}(1 - 1/n) = \inf \set{ y \; | \; 1 - F(y) \le 1/n }$ and $a_n = f(b_n)$
  [dHF, Theorem~1.1.8], [EKM, Theorem~3.3.26], [Resnick, Proposition~1.1].
\end{theorem}
\begin{proof}
  \leanok
  Apply Lemma~\ref{lem:gumbel-quantile} to $g = 1 - F$ and $c = 1/n$: the quantiles $b_n$
  satisfy $g(b_n) \le 1/n < g(y)$ for $y < b_n$, and $b_n \uparrow x_F$ since $g > 0$ to the left
  of $x_F$ (and, for finite $x_F$, $g(x_F) = 0$ forces $b_n \le x_F$, while one checks from the
  auxiliary relation that $b_n < x_F$ eventually). Lemma~\ref{lem:gumbel-sufficiency-core}, along
  $L = \texttt{atTop}$ or $L = \mathcal{N}_{<}(x_F)$, gives
  $n (1 - F(a_n x + b_n)) \to e^{-x}$, and Lemma~\ref{lem:mda-gumbel-of-tendsto-mul} concludes.
\end{proof}

\begin{theorem}[The von Mises condition]
  \label{thm:von-mises}
  \uses{thm:mda-gumbel-of-auxiliary}
  \lean{CumulativeDistributionFunction.tendsto_auxiliary_of_vonMises,
    CumulativeDistributionFunction.isInMDAGumbel_of_vonMises}
  \leanok
  Suppose $F$ is twice differentiable on $(z, \infty)$ with $F' > 0$ there, and
  \begin{align*}
    \lim_{t \to \infty} \frac{(1 - F(t)) \, F''(t)}{F'(t)^2} \; = \; -1 .
  \end{align*}
  Then $F < 1$ everywhere, $f = (1 - F)/F'$ is an auxiliary function, and hence $F$ is in the
  Gumbel domain of attraction [dHF, Theorem~1.1.8], [EKM, Theorem~3.3.27],
  [Resnick, Proposition~1.1].
\end{theorem}
\begin{proof}
  \leanok
  Since $F$ is strictly increasing on $(z, \infty)$, $F < 1$ everywhere. The function
  $f = (1 - F)/F'$ has derivative $f' = -1 - (1-F) F''/(F')^2 \to 0$. By the mean value theorem,
  $|f(t + s f(t)) - f(t)| \le |s| f(t) \sup |f'|$ over the segment, so
  $f(t + s f(t))/f(t) \to 1$ uniformly for $s$ in compact sets. With $g = 1 - F$ we have
  $(\log g)' = -1/f$, hence
  \begin{align*}
    \log \frac{g(t + x f(t))}{g(t)} \; = \; - \int_0^x \frac{f(t)}{f(t + s f(t))} \, \ud s
      \; \to \; -x .
  \end{align*}
  Theorem~\ref{thm:mda-gumbel-of-auxiliary} concludes.
\end{proof}

\begin{theorem}[The von Mises condition, finite right endpoint]
  \label{thm:von-mises-finite}
  \uses{thm:mda-gumbel-of-auxiliary, thm:von-mises}
  \lean{CumulativeDistributionFunction.tendsto_auxiliary_of_vonMises_nhdsLT,
    CumulativeDistributionFunction.isInMDAGumbel_of_vonMises_nhdsLT}
  \leanok
  Let $z < x_F$. Suppose $F$ is twice differentiable on $(z, x_F)$ with $F' > 0$ there,
  $F(t) \to 1$ as $t \uparrow x_F$, and
  \begin{align*}
    \lim_{t \uparrow x_F} \frac{(1 - F(t)) \, F''(t)}{F'(t)^2} \; = \; -1 .
  \end{align*}
  Then $x_F$ is the right endpoint of $F$, $f = (1 - F)/F'$ is an auxiliary function as
  $t \uparrow x_F$, and $F$ is in the Gumbel domain of attraction [EKM, Theorem~3.3.27],
  [dHF, Theorem~1.1.8]. The hypothesis $F(t) \to 1$ cannot be dropped: $1 - F(t) = e^{-t}$ on
  $(0, 1)$ with a jump to $1$ at $1$ satisfies the other conditions but is not in the domain.
\end{theorem}
\begin{proof}
  \leanok
  The argument of Theorem~\ref{thm:von-mises} applies on $(z, x_F)$, once we know that
  $t + x f(t) < x_F$ eventually. Near $x_F$ we have $f(t) \le \varepsilon (x_F - t)$: otherwise
  $(\log(1-F))' = -1/f$ would be bounded on $(t, x_F)$, so $\log(1 - F)$ would stay bounded
  below, contradicting $F(t) \to 1$.
\end{proof}

\begin{proposition}[A finite-endpoint example]
  \label{prop:exp-neg-inv-gumbel}
  \uses{thm:von-mises-finite}
  \lean{CumulativeDistributionFunction.expNegInvCDF,
    CumulativeDistributionFunction.isInMDAGumbel_expNegInvCDF}
  \leanok
  The c.d.f.\ $F(x) = 1 - \exp(-x/(1-x))$ on $[0, 1)$ is in the Gumbel domain of attraction,
  with right endpoint $x_F = 1$.
\end{proposition}
\begin{proof}
  \leanok
  Here $(1-F)F''/(F')^2 = 2(1-x) - 1 \to -1$ as $x \uparrow 1$.
\end{proof}

\begin{definition}[Standard exponential distribution]
  \label{def:std-exponential-cdf}
  \uses{def:cdf}
  \lean{CumulativeDistributionFunction.standardExponentialCDF,
    CumulativeDistributionFunction.standardExponentialCDF_apply_of_nonneg,
    CumulativeDistributionFunction.standardExponentialCDF_apply_of_nonpos}
  \leanok
  The standard exponential c.d.f.\ is $F(x) = 1 - e^{-x}$ for $x \ge 0$ and $F(x) = 0$ for
  $x \le 0$.
\end{definition}

\begin{proposition}[Examples of the Gumbel domain of attraction]
  \label{prop:mda-gumbel-examples}
  \uses{def:std-exponential-cdf, def:std-Gumbel-cdf, thm:von-mises, def:mda-gumbel}
  \lean{CumulativeDistributionFunction.isInMDAGumbel_standardExponentialCDF,
    CumulativeDistributionFunction.isInMDAGumbel_standardGumbelCDF}
  \leanok
  The standard exponential distribution and the standard Gumbel distribution $\GumbelCDF$ are in
  the Gumbel domain of attraction.
\end{proposition}
\begin{proof}
  \leanok
  For the exponential distribution, $F' = e^{-x}$, $F'' = -e^{-x}$ and
  $(1 - F) F''/(F')^2 = -1$ on $(0, \infty)$, so Theorem~\ref{thm:von-mises} applies. For
  $\GumbelCDF$, take $a_n = 1$, $b_n = \log n$: then $\GumbelCDF(x + \log n)^n =
  \exp(-n \, e^{-x}/n) = \GumbelCDF(x)$ for $n \ge 1$.
\end{proof}

\begin{lemma}[Comparison of levels]
  \label{lem:gumbel-eventually-lt}
  \lean{ExtremeValueTheory.Gumbel.eventually_lt_of_tendsto_natCast_mul,
    ExtremeValueTheory.Gumbel.eventually_lt_of_tendsto_natCast_mul_one}
  \leanok
  Let $g$ be antitone. If $n \, g(p_n) \to \mu$ and $n \, g(q_n) \to \nu$ with $\nu < \mu$, then
  eventually $p_n < q_n$. If $n \, g(b_n) \to 1$ and $g(t) > 0$, then eventually $t < b_n$.
\end{lemma}
\begin{proof}
  \leanok
  If $q_n \le p_n$ then $n \, g(p_n) \le n \, g(q_n)$, which is eventually impossible. For the
  second claim, $b_n \le t$ would give $n \, g(b_n) \ge n \, g(t) \to \infty$.
\end{proof}

\begin{lemma}[Necessity, analytic core]
  \label{lem:gumbel-necessity-core}
  \uses{lem:gumbel-eventually-lt}
  \lean{ExtremeValueTheory.Gumbel.exists_auxiliary_of_tendsto_mul}
  \leanok
  Let $g$ be antitone, $a_n > 0$, and $n \, g(a_n x + b_n) \to e^{-x}$ for every $x$. Let $L$ be a
  filter on $\bR$ (describing $t \uparrow x_F$) such that, for $t$ eventually in $L$, eventually
  $t < b_n$, and such that for every $K$, eventually in $L$ there is $n \ge K$ with $b_n \le t$.
  Then there is a positive function $f$ with $g(t + x f(t))/g(t) \to e^{-x}$ along $L$, for
  every $x$.
\end{lemma}
\begin{proof}
  \leanok
  We avoid the classical route through $\Pi$-variation of the quantile function and use a direct
  sandwich argument. Comparing the levels $n$ and $n+1$ (Lemma~\ref{lem:gumbel-eventually-lt},
  using $n \, g(b_{n+1}) \to 1 > e^{-\delta}$) gives, for every $\delta > 0$, eventually
  $b_{n+1} < b_n + \delta a_n$. For $t$ in $L$ let $N(t) = \sup \set{ n \; | \; b_n \le t }$,
  which is finite by the first hypothesis, so that $b_{N(t)} \le t < b_{N(t)+1}$, and
  $N(t) \to \infty$ along $L$ by the second hypothesis. Put $f(t) = a_{N(t)}$ and $n = N(t)$.
  Then $b_n \le t < b_n + \delta a_n$, so by monotonicity
  \begin{align*}
    \frac{g(a_n (x + \delta) + b_n)}{g(b_n)} \; \le \; \frac{g(t + x f(t))}{g(t)}
      \; \le \; \frac{g(a_n x + b_n)}{g(a_n \delta + b_n)} ,
  \end{align*}
  and the outer ratios tend to $e^{-x - \delta}$ and $e^{-x + \delta}$. Letting $\delta \to 0$
  concludes.
\end{proof}

\begin{theorem}[Necessity of the auxiliary function]
  \label{thm:gumbel-necessity}
  \uses{lem:mda-gumbel-of-tendsto-mul, lem:gumbel-necessity-core, lem:gumbel-eventually-lt}
  \lean{CumulativeDistributionFunction.exists_auxiliary_atTop_of_isInMDAGumbel,
    CumulativeDistributionFunction.exists_auxiliary_nhdsLT_of_isInMDAGumbel}
  \leanok
  If $F$ is in the Gumbel domain of attraction, then $F$ admits a positive auxiliary function,
  both when $x_F = +\infty$ and when $x_F < \infty$ (de Haan 1970; [dHF, Theorem~1.2.5],
  [Resnick, Proposition~1.4], [EKM, Theorem~3.3.26]).
\end{theorem}
\begin{proof}
  \leanok
  Lemma~\ref{lem:mda-gumbel-of-tendsto-mul} gives $a_n > 0$, $b_n$ with
  $n (1 - F(a_n x + b_n)) \to e^{-x}$; in particular $n (1 - F(b_n)) \to 1$. Apply
  Lemma~\ref{lem:gumbel-necessity-core} with $g = 1 - F$. Its first hypothesis follows from
  Lemma~\ref{lem:gumbel-eventually-lt} since $1 - F(t) > 0$ for $t < x_F$. For the second: if
  $x_F = \infty$, $b_K \le t$ for all large $t$; if $x_F < \infty$, choose $n \ge K$ with
  $n (1 - F(b_n)) > 0$, so $b_n < x_F$, and then $b_n \le t$ for $t \in (b_n, x_F)$.
\end{proof}

\begin{lemma}[Finite right endpoint]
  \label{lem:right-endpoint}
  \uses{lem:gumbel-quantile}
  \lean{CumulativeDistributionFunction.exists_rightEndpoint_of_not_lt_one}
  \leanok
  If $F(x_1) = 1$ for some $x_1$, then there is $x_F \in \bR$ with $F(x) < 1$ for $x < x_F$ and
  $F(x_F) = 1$.
\end{lemma}
\begin{proof}
  \leanok
  Take $x_F = \inf \set{ x \; | \; 1 - F(x) \le 0 }$ and apply Lemma~\ref{lem:gumbel-quantile}
  with $c = 0$.
\end{proof}

\begin{theorem}[De Haan's characterization of the Gumbel domain of attraction]
  \label{thm:mda-gumbel-iff}
  \uses{thm:mda-gumbel-of-auxiliary, thm:gumbel-necessity, lem:right-endpoint}
  \lean{CumulativeDistributionFunction.isInMDAGumbel_iff_atTop,
    CumulativeDistributionFunction.isInMDAGumbel_iff_nhdsLT,
    CumulativeDistributionFunction.isInMDAGumbel_iff}
  \leanok
  A c.d.f.\ $F$ is in the Gumbel domain of attraction if and only if it admits an auxiliary
  function. More precisely, either $F < 1$ everywhere and there is $f$, eventually positive, with
  $(1 - F(t + x f(t)))/(1 - F(t)) \to e^{-x}$ as $t \to \infty$; or $F$ has a finite right
  endpoint $x_F$ and there is $f$, positive near $x_F$, with the same relation as $t \uparrow x_F$
  [dHF, Theorems~1.1.8 and~1.2.5], [EKM, Theorem~3.3.26], [Resnick, Propositions~1.1 and~1.4].
\end{theorem}
\begin{proof}
  \leanok
  Combine Theorems~\ref{thm:mda-gumbel-of-auxiliary} and~\ref{thm:gumbel-necessity}; the case
  split on whether $F$ reaches $1$ uses Lemma~\ref{lem:right-endpoint}.
\end{proof}

\section{Von Mises conditions for the Fr\'echet and Weibull domains}

\begin{theorem}[Von Mises condition, Fr\'echet]
  \label{thm:von-mises-frechet}
  \uses{thm:mda-frechet-iff, thm:karamata-converse}
  \lean{CumulativeDistributionFunction.isRegularlyVarying_of_vonMises, CumulativeDistributionFunction.isInMDAFrechet_of_vonMises}
  \leanok
  If $F$ is differentiable on $(z, \infty)$ and $x F'(x)/(1 - F(x)) \to \alpha > 0$, then
  $1 - F \in RV_{-\alpha}$ and $F$ is in the Fr\'echet domain of attraction
  [EKM, Corollary~3.3.8], [dHF, Theorem~1.1.8].
\end{theorem}
\begin{proof}
  \leanok
  $(\log(1-F))'(x) = -F'(x)/(1-F(x))$, so $x (\log(1 - F))' \to -\alpha$; apply
  Theorem~\ref{thm:karamata-converse} and Theorem~\ref{thm:mda-frechet-iff}.
\end{proof}

\begin{theorem}[Von Mises condition, Weibull]
  \label{thm:von-mises-weibull}
  \uses{thm:mda-weibull-iff, thm:karamata-converse}
  \lean{CumulativeDistributionFunction.isRegularlyVarying_of_vonMises_nhdsLT, CumulativeDistributionFunction.isInMDAWeibull_of_vonMises}
  \leanok
  If $F$ is differentiable on $(z, x_F)$ and $(x_F - x) F'(x)/(1 - F(x)) \to \alpha > 0$ as
  $x \uparrow x_F$, then $x_F$ is the right endpoint of $F$, $t \mapsto 1 - F(x_F - 1/t)$ is in
  $RV_{-\alpha}$, and $F$ is in the Weibull domain of attraction [EKM, Corollary~3.3.13].
\end{theorem}
\begin{proof}
  \leanok
  Substitute $x = x_F - 1/t$ and argue as in Theorem~\ref{thm:von-mises-frechet}.
\end{proof}

\begin{proposition}[Pareto and power-law examples]
  \label{prop:von-mises-examples}
  \uses{thm:von-mises-frechet, thm:von-mises-weibull}
  \lean{CumulativeDistributionFunction.paretoCDF, CumulativeDistributionFunction.isInMDAFrechet_paretoCDF, CumulativeDistributionFunction.powerCDF, CumulativeDistributionFunction.isInMDAWeibull_powerCDF}
  \leanok
  The Pareto law $1 - x^{-\alpha}$ on $[1, \infty)$ is in the Fr\'echet domain, and the law
  $1 - (1 - x)^\alpha$ on $[0, 1]$ (uniform for $\alpha = 1$, Beta$(1, \alpha)$ in general) is in
  the Weibull domain.
\end{proposition}
\begin{proof}
  \leanok
  The von Mises ratios are identically $\alpha$.
\end{proof}
