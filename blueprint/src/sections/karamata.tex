\chapter{Karamata's theorem}

Let $F \colon \bR \to \bR$ be antitone and regularly varying with index $-\alpha$, where
$\alpha > 1$. Typically $F = 1 - G$ is the survival function of a c.d.f.\ $G$ in the
Fréchet domain of attraction (Chapter~\ref{chap:domains-of-attraction}).

\begin{lemma}[Positivity of antitone eventually positive functions]
  \label{lem:pos-of-antitone}
  \lean{ExtremeValueTheory.IsRegularlyVarying.pos_of_antitone}
  \leanok
  An antitone function $F$ with $F(x) > 0$ for all sufficiently large $x$ is positive everywhere.
\end{lemma}
\begin{proof}
  \leanok
  For any $t$ there is $x \ge t$ with $F(x) > 0$, and $F(t) \ge F(x)$.
\end{proof}

\begin{lemma}[Integrability of regularly varying tails]
  \label{lem:rv-integrableOn-Ioi}
  \uses{def:regularly-varying, thm:rv-potter-monotone}
  \lean{ExtremeValueTheory.IsRegularlyVarying.integrableOn_Ioi}
  \leanok
  If $F$ is antitone and regularly varying with index $-\alpha$, $\alpha > 1$, then $F$ is
  integrable on every half-line $(x, \infty)$.
\end{lemma}
\begin{proof}
  \leanok
  By the Potter bounds (Theorem~\ref{thm:rv-potter-monotone}) with
  $\delta = (\alpha - 1)/2$, $F(t) \le C t^{-\alpha+\delta}$ for $t$ large, and
  $-\alpha + \delta < -1$, so $F$ is integrable near $+\infty$. On bounded intervals
  $F$ is bounded and measurable, being monotone.
\end{proof}

\begin{theorem}[Karamata's theorem for tail integrals]
  \label{thm:karamata}
  \uses{def:regularly-varying, thm:rv-potter-monotone, lem:rv-integrableOn-Ioi, lem:pos-of-antitone}
  \lean{ExtremeValueTheory.IsRegularlyVarying.tendsto_integral_Ioi_div}
  \leanok
  Let $F$ be antitone and regularly varying with index $-\alpha$, where $\alpha > 1$. Then
  \begin{align*}
    \lim_{x \to \infty} \frac{\int_x^\infty F(t) \, \ud t}{x \, F(x)} \; = \; \frac{1}{\alpha - 1}
  \end{align*}
  [BGT, Proposition~1.5.10; see also Theorem~1.5.11]. In particular, if $F$ is the survival
  function of a random variable, the mean excess over a high threshold $x$ is asymptotically
  $x/(\alpha - 1)$.
\end{theorem}
\begin{proof}
  \leanok
  After the substitution $t = x u$,
  \begin{align*}
    \frac{\int_x^\infty F(t) \, \ud t}{x \, F(x)} \; = \; \int_1^\infty \frac{F(x u)}{F(x)} \, \ud u .
  \end{align*}
  The integrand tends to $u^{-\alpha}$ pointwise. By the Potter bounds
  (Theorem~\ref{thm:rv-potter-monotone}) with $\delta = (\alpha-1)/2$ and $A = 2$, for $x$
  large it is dominated by $2 u^{-\alpha + \delta}$, which is integrable on $(1, \infty)$.
  Dominated convergence yields the limit $\int_1^\infty u^{-\alpha} \, \ud u = \frac{1}{\alpha - 1}$.
\end{proof}

\begin{theorem}[Karamata's theorem, relaxed hypotheses]
  \label{thm:karamata-relaxed}
  \uses{thm:karamata}
  \lean{ExtremeValueTheory.IsRegularlyVarying.tendsto_integral_Ioi_div_of_measurable,
    ExtremeValueTheory.IsRegularlyVarying.tendsto_integral_Ioi_div_of_antitoneOn}
  \leanok
  The conclusion of Theorem~\ref{thm:karamata} holds for every measurable $F \in RV_{-\alpha}$,
  $\alpha > 1$, and for every $F \in RV_{-\alpha}$ that is antitone on some $[a, \infty)$
  [BGT, Proposition~1.5.10].
\end{theorem}
\begin{proof}
  \leanok
  Potter bounds for measurable regularly varying functions give the domination needed in the
  proof of Theorem~\ref{thm:karamata}.
\end{proof}

\begin{theorem}[Karamata's theorem, converse half]
  \label{thm:karamata-converse}
  \uses{thm:karamata-representation}
  \lean{ExtremeValueTheory.IsRegularlyVarying.of_tendsto_mul_div_integral_Ioi_of_pos, ExtremeValueTheory.IsRegularlyVarying.of_log_sub_eq_integral,
    ExtremeValueTheory.IsRegularlyVarying.of_tendsto_mul_deriv_div, ExtremeValueTheory.IsRegularlyVarying.of_abs_log_sub_le, ExtremeValueTheory.IsRegularlyVarying.of_tendsto}
  \leanok
  Let $F$ be antitone and integrable on $(a, \infty)$, and $\alpha > 1$. If
  $x F(x) / \int_x^\infty F \to \alpha - 1$, then $F \in RV_{-\alpha}$ [BGT, Theorem~1.6.1].
  Similarly, if $g > 0$ is differentiable with $x g'(x)/g(x) \to \rho$, then $g \in RV_\rho$.
\end{theorem}
\begin{proof}
  \leanok
  With $I(x) = \int_x^\infty F$, $(\log I)'(x) = -F(x)/I(x)$, so $x (\log I)'(x) \to 1 - \alpha$.
  The representation theorem gives $I \in RV_{1-\alpha}$, and $F = (I/x) \cdot (x F/I)$.
\end{proof}

\begin{theorem}[{Karamata's theorem for integrals over $[a, x]$}]
  \label{thm:karamata-zero}
  \uses{thm:karamata-relaxed}
  \lean{ExtremeValueTheory.IsRegularlyVarying.tendsto_integral_interval_div}
  \leanok
  If $f \in RV_\rho$ is measurable and integrable on $[a, x]$ for $x \ge a$, and $\rho > -1$, then
  $\int_a^x f / (x f(x)) \to 1/(\rho + 1)$ [BGT, Proposition~1.5.8].
\end{theorem}
\begin{proof}
  \leanok
  Uniform convergence on compacts and Potter bounds near $0$ after rescaling $t = x s$.
\end{proof}

\section{Karamata's theorem without monotonicity}

\begin{theorem}[Karamata's theorem, measurable near infinity]
  \label{thm:karamata-near-top}
  \uses{thm:karamata-relaxed, thm:karamata-zero, thm:rv-uct-near-top}
  \lean{ExtremeValueTheory.IsRegularlyVarying.tendsto_integral_mul_mul_div,
    ExtremeValueTheory.IsRegularlyVarying.tendsto_integral_Ioi_div_of_measurableNearTop,
    ExtremeValueTheory.IsRegularlyVarying.tendsto_integral_div_of_measurableNearTop,
    ExtremeValueTheory.IsRegularlyVarying.eventually_tendsto_integral_div_of_measurableNearTop,
    ExtremeValueTheory.IsRegularlyVarying.exists_integrableOn_Ioi_of_measurableNearTop}
  \leanok
  If $f \in RV_\rho$ is measurable on some $[X, \infty)$, then
  $\int_{px}^{qx} f / (x f(x)) \to \int_p^q u^\rho \, du$ for $p, q > 0$. For $\rho > -1$,
  $\int_a^x f / (x f(x)) \to 1/(\rho + 1)$, for every $a$ with $f$ integrable on $[a, x]$, and for every
  large $a$ with no integrability hypothesis. For $\rho < -1$, $\int_x^\infty f / (x f(x)) \to -1/(\rho + 1)$
  [BGT, Propositions~1.5.8, 1.5.10].
\end{theorem}
\begin{proof}
  \leanok
  Replace $f$ by $f \mathbf{1}_{[X, \infty)}$, which changes neither the index nor the limits, and apply
  the measurable forms.
\end{proof}

\begin{theorem}[Karamata's theorem at the index $-1$]
  \label{thm:karamata-neg-one}
  \uses{thm:karamata-near-top}
  \lean{ExtremeValueTheory.IsRegularlyVarying.tendsto_integral_div_atTop_of_neg_one,
    ExtremeValueTheory.IsRegularlyVarying.tendsto_integral_Ioi_div_atTop_of_neg_one}
  \leanok
  If $f \in RV_{-1}$ is measurable near infinity, then $\int_a^x f / (x f(x)) \to \infty$ when $f \ge 0$ on
  $[a, \infty)$ and integrable on $[a, x]$, and $\int_x^\infty f / (x f(x)) \to \infty$ when $f$ is integrable
  on some half-line [BGT, Proposition~1.5.9].
\end{theorem}
\begin{proof}
  \leanok
  For each $M$, $\int_{x/M}^{x} f / (x f(x)) \to \log M$, and $f \ge 0$ eventually.
\end{proof}

\begin{theorem}[Integrals of regularly varying functions are regularly varying]
  \label{thm:rv-integrals}
  \uses{thm:karamata-near-top, thm:karamata-neg-one}
  \lean{ExtremeValueTheory.IsRegularlyVarying.integral, ExtremeValueTheory.IsRegularlyVarying.integral_of_neg_one,
    ExtremeValueTheory.IsRegularlyVarying.integral_Ioi_of_measurableNearTop,
    ExtremeValueTheory.IsRegularlyVarying.integral_Ioi_of_neg_one}
  \leanok
  With $f \in RV_\rho$ measurable near infinity: $x \mapsto \int_a^x f \in RV_{\rho + 1}$ for $\rho > -1$,
  $x \mapsto \int_x^\infty f \in RV_{\rho + 1}$ for $\rho < -1$, and both are slowly varying at $\rho = -1$
  [BGT, Propositions~1.5.8--1.5.10].
\end{theorem}
\begin{proof}
  \leanok
  The integral is asymptotic to a regularly varying function, by Karamata's theorem. At $\rho = -1$, the
  ratio of the integrals at $x$ and $cx$ differs from $1$ by a term that tends to $0$.
\end{proof}

\begin{theorem}[Karamata's converse with weights]
  \label{thm:karamata-converse-weighted}
  \uses{thm:karamata-converse}
  \lean{ExtremeValueTheory.IsRegularlyVarying.of_tendsto_mul_div_integral,
    ExtremeValueTheory.IsRegularlyVarying.of_tendsto_rpow_mul_div_integral,
    ExtremeValueTheory.IsRegularlyVarying.of_tendsto_rpow_mul_div_integral_Ioi}
  \leanok
  If $f$ is eventually positive and $x^{\sigma + 1} f(x) / \int_a^x t^\sigma f \to \sigma + \rho + 1 > 0$, or
  $x^{\sigma + 1} f(x) / \int_x^\infty t^\sigma f \to -(\sigma + \rho + 1) > 0$, then $f \in RV_\rho$
  [BGT, Theorem~1.6.1].
\end{theorem}
\begin{proof}
  \leanok
  A nonzero limit forces integrability, since a non-integrable integrand gives the value $0$. The
  logarithm of the integral then has the logarithmic derivative that the representation theorem needs.
\end{proof}
