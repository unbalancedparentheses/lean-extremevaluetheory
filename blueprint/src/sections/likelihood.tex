\chapter{Likelihood inference for the GPD and the GEV}
\label{chap:likelihood}

We follow Smith, \emph{Maximum likelihood estimation in a class of nonregular cases}, Biometrika 72
(1985), Coles (2001), Ch.~3--4, and [dHF, \S3.4].

\begin{theorem}[Densities of the GPD and GEV]
  \label{thm:gpd-gev-density}
  \uses{def:gev}
  \lean{ExtremeValueTheory.eq_withDensity_of_hasDerivAt_off, ExtremeValueTheory.integral_eq_one_of_eq_withDensity,
    ExtremeValueTheory.gpdMeasure, ExtremeValueTheory.gpdMeasure_Ioi, ExtremeValueTheory.gpdMeasure_Iic,
    ExtremeValueTheory.hasDerivAt_gpdCDF, ExtremeValueTheory.gpdMeasure_eq_withDensity,
    ExtremeValueTheory.integral_gpdDensity, ExtremeValueTheory.gpdLogLik, ExtremeValueTheory.log_gpdDensity,
    ExtremeValueTheory.continuous_gevFun, ExtremeValueTheory.hasDerivAt_gevFun,
    ExtremeValueTheory.measure_gevCDF_eq_withDensity, ExtremeValueTheory.integral_gevDensity,
    ExtremeValueTheory.gevLogLik}
  \leanok
  The GPD law (that of $\sigma(e^{\xi E} - 1)/\xi$ with $E$ standard exponential) has survival function $\bar G_{\xi,\sigma}$ and
  density $\sigma^{-1}(1 + \xi x/\sigma)^{-1/\xi - 1}$ on its support; the GEV law has the density obtained by
  differentiating $H_\xi$; both integrate to $1$.
\end{theorem}
\begin{proof}
  \leanok
  A measure whose continuous distribution function is differentiable off finitely many points with derivative
  $f \ge 0$ is $f \ud x$.
\end{proof}

\begin{theorem}[Exponential case]
  \label{thm:mle-exponential}
  \uses{thm:gpd-gev-density}
  \lean{ExtremeValueTheory.gpdLogLik_zero_lt_sampleMean, ExtremeValueTheory.isMaxOn_gpdLogLik_zero,
    ExtremeValueTheory.ae_tendsto_sampleMean_gpd_zero, ExtremeValueTheory.tendstoInDistribution_sampleMean_gpd_zero}
  \leanok
  For $\xi = 0$ the sample mean is the unique MLE of $\sigma$; it is strongly consistent and
  $\sqrt n(\hat\sigma_n - \sigma) \Rightarrow N(0, \sigma^2)$.
\end{theorem}
\begin{proof}
  \leanok
  Mathlib's strong law and central limit theorem.
\end{proof}

\begin{theorem}[Scale MLE with known shape]
  \label{thm:mle-scale}
  \uses{thm:gpd-gev-density}
  \lean{ExtremeValueTheory.hasDerivAt_gpdLogLik, ExtremeValueTheory.exists_gpdScaleScore_eq,
    ExtremeValueTheory.gpdScaleScore_root_unique, ExtremeValueTheory.gpdLogLik_lt_of_score_eq,
    ExtremeValueTheory.ae_existsUnique_gpdScaleMLE, ExtremeValueTheory.ae_tendsto_gpdScaleMLE}
  \leanok
  For known $\xi > -1$, $\xi \ne 0$, the score equation in $\sigma$ has a unique admissible root for positive data, it is the
  unique maximizer of the likelihood, and it is strongly consistent for iid GPD data.
\end{theorem}
\begin{proof}
  \leanok
  The score $\sigma^{-1}((1 + \xi)\sum_i x_i/(\sigma + \xi x_i) - n)$ changes sign exactly once.
\end{proof}

\begin{theorem}[Wald consistency and the joint GPD MLE]
  \label{thm:wald}
  \uses{thm:gpd-gev-density}
  \lean{ExtremeValueTheory.ae_tendsto_of_wald, ExtremeValueTheory.gpd_params_eq_of_gpdMeasure_eq,
    ExtremeValueTheory.integral_log_gpdDensity_lt, ExtremeValueTheory.exists_isMaxOn_gpdLogLik,
    ExtremeValueTheory.ae_tendsto_gpdMLE}
  \leanok
  On a compact metric parameter space, a criterion continuous in the parameter with an integrable envelope and a
  unique maximizer of its expectation yields strongly consistent approximate maximizers [Wald 1949]. For the GPD
  this gives strong consistency of the joint MLE of $(\xi, \sigma)$ on compact sets in $\{\xi > 0, \sigma > 0\}$. The range
  $-1/2 < \xi \le 0$ (parameter-dependent support), asymptotic normality for $\xi \ne 0$, and the GEV MLE are not formalized.
\end{theorem}
\begin{proof}
  \leanok
  Uniform strong law over the compact set via the envelope; identifiability and the Kullback--Leibler inequality.
\end{proof}

\begin{theorem}[Extended-real Wald consistency and the joint GPD MLE for all shapes]
  \label{thm:wald-ereal}
  \uses{thm:wald}
  \lean{ExtremeValueTheory.ae_tendsto_of_wald_ereal, ExtremeValueTheory.ae_tendsto_of_wald_ereal_of_continuous, ExtremeValueTheory.exists_integral_toReal_max_lt,
    ExtremeValueTheory.gpdLogDensityE, ExtremeValueTheory.gpdLogLikE, ExtremeValueTheory.continuousOn_gpdLogDensityE, ExtremeValueTheory.gpd_params_eq_of_gpdMeasure_eq',
    ExtremeValueTheory.integral_log_lt_of_density, ExtremeValueTheory.exists_isMaxOn_gpdLogLikE, ExtremeValueTheory.ae_tendsto_gpdMLE'}
  \leanok
  Wald's theorem holds for criteria with values in $[-\infty, \infty)$, upper semicontinuous in the parameter, with an
  integrable upper envelope [Wald 1949]. With the log-likelihood set to $-\infty$ off the support, this gives strong
  consistency of the joint MLE of $(\xi, \sigma)$ on every compact subset of $\{\xi > -1, \sigma > 0\}$, including $\xi \le 0$ where
  the support depends on the parameters [Smith 1985].
\end{theorem}
\begin{proof}
  \leanok
  On the support, $\log g = -\log\sigma - (1+\xi)(x/\sigma)\,\lambda(\xi x/\sigma)$ with $\lambda(t) = \log(1+t)/t$ continuous on $(-1,\infty)$, which
  gives continuity across $\xi = 0$; a Kullback--Leibler inequality allowing different supports gives identifiability.
\end{proof}

\begin{theorem}[Asymptotic normality of the scale MLE]
  \label{thm:mle-scale-normality}
  \uses{thm:mle-scale}
  \lean{ExtremeValueTheory.tendstoInDistribution_gpdScaleMLE}
  \leanok
  For known $\xi > -1/2$, $\xi \ne 0$, a measurable sequence of likelihood roots satisfies
  $\sqrt n(\hat\sigma_n - \sigma_0) \Rightarrow N(0, \sigma_0^2(1 + 2\xi))$, the inverse Fisher information [Smith 1985]. The joint
  $(\xi, \sigma)$ normality and GEV likelihood theory are not formalized.
\end{theorem}
\begin{proof}
  \leanok
  An exact identity $F_n(\sigma_0) = (\hat\sigma_n - \sigma_0) R_n$ for the score, the CLT for the score (variance needs
  $\xi > -1/2$), $R_n \to 1/(\sigma_0(1+2\xi))$ in probability, and Slutsky; for $\xi < 0$ a tail estimate controls
  $\max_i 1/(\sigma_0 + \xi X_i) = o_P(\sqrt n)$.
\end{proof}

\begin{theorem}[A measurable scale MLE]
  \label{thm:gpd-scale-mle-measurable}
  \uses{thm:mle-scale-normality}
  \lean{ExtremeValueTheory.gpdScaleMLE, ExtremeValueTheory.measurable_gpdScaleMLE, ExtremeValueTheory.gpdScaleMLE_eq_of_root, ExtremeValueTheory.ae_gpdScaleMLE_isMLE,
    ExtremeValueTheory.ae_tendsto_gpdScaleMLE', ExtremeValueTheory.tendstoInDistribution_gpdScaleMLE'}
  \leanok
  The supremum over rational admissible scales with nonnegative score is a measurable function of the sample that is
  almost surely the unique root and the unique likelihood maximizer; it is strongly consistent and
  $\sqrt n(\hat\sigma_n - \sigma_0) \Rightarrow N(0, \sigma_0^2(1+2\xi))$ for $\xi > -1/2$, $\xi \ne 0$, with no further hypotheses.
\end{theorem}
\begin{proof}
  \leanok
  The score is strictly decreasing in $\sigma$ and the admissible set is open.
\end{proof}

\begin{theorem}[Fisher information and joint asymptotic normality]
  \label{thm:gpd-joint-normality}
  \uses{thm:wald-ereal, thm:gpd-scale-mle-measurable}
  \lean{ExtremeValueTheory.inv_gpdFisherInfo, ExtremeValueTheory.tendstoInDistribution_gpdScoreVec, ExtremeValueTheory.tendstoInDistribution_gpdJointMLE,
    ExtremeValueTheory.tendstoInDistribution_gpdMLE, ExtremeValueTheory.score_eq_zero_of_isMaxOn}
  \leanok
  For $\xi > -1/2$, $\xi \ne 0$, the GPD score has mean zero and covariance
  $I(\xi,\sigma) = \frac{1}{(1+\xi)(1+2\xi)}\begin{pmatrix} 2 & 1/\sigma \ 1/\sigma & (1+\xi)/\sigma^2 \end{pmatrix}$, equal to minus the expected
  Hessian, with $I^{-1} = (1+\xi)\begin{pmatrix} 1+\xi & -\sigma \ -\sigma & 2\sigma^2 \end{pmatrix}$, and the normalized score is
  asymptotically $N(0, I)$. For $\xi_0 > 0$, measurable maximizers over a compact set with $\theta_0$ in its interior satisfy
  $\sqrt n(\hat\theta_n - \theta_0) \Rightarrow N(0, I(\theta_0)^{-1})$ [Smith 1985], [Coles, \S4.3]. The joint result for
  $-1/2 < \xi_0 < 0$ (moving support) is not formalized.
\end{theorem}
\begin{proof}
  \leanok
  An exact integral form of the mean-value expansion, a Lipschitz bound on the Hessian near $\theta_0$ (uses $\xi_0 > 0$), the strong
  law for the averaged observed information, Cram\'er--Wold for the score, and Slutsky.
\end{proof}
