\chapter{Likelihood inference for the GPD and the GEV}
\label{chap:likelihood}

We follow Smith, \emph{Maximum likelihood estimation in a class of nonregular cases}, Biometrika 72
(1985), Coles (2001), Ch.~3--4, and [dHF, \S3.4].

\begin{theorem}[Densities of the GPD and GEV]
  \label{thm:gpd-gev-density}
  \uses{def:gev}
  \lean{ExtremeValueTheory.eq_withDensity_of_hasDerivAt_off, ExtremeValueTheory.integral_eq_one_of_eq_withDensity,
    ExtremeValueTheory.gpdMeasure, ExtremeValueTheory.gpdMeasure_Ioi, ExtremeValueTheory.gpdMeasure_Iic,
    ExtremeValueTheory.hasDerivAt_gpdCDF, ExtremeValueTheory.gpdMeasure_eq_withDensity,
    ExtremeValueTheory.integral_gpdDensity, ExtremeValueTheory.gpdLogLik, ExtremeValueTheory.log_gpdDensity,
    ExtremeValueTheory.continuous_gevFun, ExtremeValueTheory.hasDerivAt_gevFun,
    ExtremeValueTheory.measure_gevCDF_eq_withDensity, ExtremeValueTheory.integral_gevDensity,
    ExtremeValueTheory.gevLogLik}
  \leanok
  The GPD law (that of $\sigma(e^{\xi E} - 1)/\xi$ with $E$ standard exponential) has survival function $\bar G_{\xi,\sigma}$ and
  density $\sigma^{-1}(1 + \xi x/\sigma)^{-1/\xi - 1}$ on its support; the GEV law has the density obtained by
  differentiating $H_\xi$; both integrate to $1$.
\end{theorem}
\begin{proof}
  \leanok
  A measure whose continuous distribution function is differentiable off finitely many points with derivative
  $f \ge 0$ is $f \ud x$.
\end{proof}

\begin{theorem}[Exponential case]
  \label{thm:mle-exponential}
  \uses{thm:gpd-gev-density}
  \lean{ExtremeValueTheory.gpdLogLik_zero_lt_sampleMean, ExtremeValueTheory.isMaxOn_gpdLogLik_zero,
    ExtremeValueTheory.ae_tendsto_sampleMean_gpd_zero, ExtremeValueTheory.tendstoInDistribution_sampleMean_gpd_zero}
  \leanok
  For $\xi = 0$ the sample mean is the unique MLE of $\sigma$; it is strongly consistent and
  $\sqrt n(\hat\sigma_n - \sigma) \Rightarrow N(0, \sigma^2)$.
\end{theorem}
\begin{proof}
  \leanok
  Mathlib's strong law and central limit theorem.
\end{proof}

\begin{theorem}[Scale MLE with known shape]
  \label{thm:mle-scale}
  \uses{thm:gpd-gev-density}
  \lean{ExtremeValueTheory.hasDerivAt_gpdLogLik, ExtremeValueTheory.exists_gpdScaleScore_eq,
    ExtremeValueTheory.gpdScaleScore_root_unique, ExtremeValueTheory.gpdLogLik_lt_of_score_eq,
    ExtremeValueTheory.ae_existsUnique_gpdScaleMLE, ExtremeValueTheory.ae_tendsto_gpdScaleMLE}
  \leanok
  For known $\xi > -1$, $\xi \ne 0$, the score equation in $\sigma$ has a unique admissible root for positive data, it is the
  unique maximizer of the likelihood, and it is strongly consistent for iid GPD data.
\end{theorem}
\begin{proof}
  \leanok
  The score $\sigma^{-1}((1 + \xi)\sum_i x_i/(\sigma + \xi x_i) - n)$ changes sign exactly once.
\end{proof}

\begin{theorem}[Wald consistency and the joint GPD MLE]
  \label{thm:wald}
  \uses{thm:gpd-gev-density}
  \lean{ExtremeValueTheory.ae_tendsto_of_wald, ExtremeValueTheory.gpd_params_eq_of_gpdMeasure_eq,
    ExtremeValueTheory.integral_log_gpdDensity_lt, ExtremeValueTheory.exists_isMaxOn_gpdLogLik,
    ExtremeValueTheory.ae_tendsto_gpdMLE}
  \leanok
  On a compact metric parameter space, a criterion continuous in the parameter with an integrable envelope and a
  unique maximizer of its expectation yields strongly consistent approximate maximizers [Wald 1949]. For the GPD
  this gives strong consistency of the joint MLE of $(\xi, \sigma)$ on compact sets in $\{\xi > 0, \sigma > 0\}$. The range
  $-1/2 < \xi \le 0$ (parameter-dependent support), asymptotic normality for $\xi \ne 0$, and the GEV MLE are not formalized.
\end{theorem}
\begin{proof}
  \leanok
  Uniform strong law over the compact set via the envelope; identifiability and the Kullback--Leibler inequality.
\end{proof}

\begin{theorem}[Extended-real Wald consistency and the joint GPD MLE for all shapes]
  \label{thm:wald-ereal}
  \uses{thm:wald}
  \lean{ExtremeValueTheory.ae_tendsto_of_wald_ereal, ExtremeValueTheory.ae_tendsto_of_wald_ereal_of_continuous, ExtremeValueTheory.exists_integral_toReal_max_lt,
    ExtremeValueTheory.gpdLogDensityE, ExtremeValueTheory.gpdLogLikE, ExtremeValueTheory.continuousOn_gpdLogDensityE, ExtremeValueTheory.gpd_params_eq_of_gpdMeasure_eq',
    ExtremeValueTheory.integral_log_lt_of_density, ExtremeValueTheory.exists_isMaxOn_gpdLogLikE, ExtremeValueTheory.ae_tendsto_gpdMLE'}
  \leanok
  Wald's theorem holds for criteria with values in $[-\infty, \infty)$, upper semicontinuous in the parameter, with an
  integrable upper envelope [Wald 1949]. With the log-likelihood set to $-\infty$ off the support, this gives strong
  consistency of the joint MLE of $(\xi, \sigma)$ on every compact subset of $\{\xi > -1, \sigma > 0\}$, including $\xi \le 0$ where
  the support depends on the parameters [Smith 1985].
\end{theorem}
\begin{proof}
  \leanok
  On the support, $\log g = -\log\sigma - (1+\xi)(x/\sigma)\,\lambda(\xi x/\sigma)$ with $\lambda(t) = \log(1+t)/t$ continuous on $(-1,\infty)$, which
  gives continuity across $\xi = 0$; a Kullback--Leibler inequality allowing different supports gives identifiability.
\end{proof}

\begin{theorem}[Asymptotic normality of the scale MLE]
  \label{thm:mle-scale-normality}
  \uses{thm:mle-scale}
  \lean{ExtremeValueTheory.tendstoInDistribution_gpdScaleMLE}
  \leanok
  For known $\xi > -1/2$, $\xi \ne 0$, a measurable sequence of likelihood roots satisfies
  $\sqrt n(\hat\sigma_n - \sigma_0) \Rightarrow N(0, \sigma_0^2(1 + 2\xi))$, the inverse Fisher information [Smith 1985]. The joint
  $(\xi, \sigma)$ normality and GEV likelihood theory are not formalized.
\end{theorem}
\begin{proof}
  \leanok
  An exact identity $F_n(\sigma_0) = (\hat\sigma_n - \sigma_0) R_n$ for the score, the CLT for the score (variance needs
  $\xi > -1/2$), $R_n \to 1/(\sigma_0(1+2\xi))$ in probability, and Slutsky; for $\xi < 0$ a tail estimate controls
  $\max_i 1/(\sigma_0 + \xi X_i) = o_P(\sqrt n)$.
\end{proof}

\begin{theorem}[A measurable scale MLE]
  \label{thm:gpd-scale-mle-measurable}
  \uses{thm:mle-scale-normality}
  \lean{ExtremeValueTheory.gpdScaleMLE, ExtremeValueTheory.measurable_gpdScaleMLE, ExtremeValueTheory.gpdScaleMLE_eq_of_root, ExtremeValueTheory.ae_gpdScaleMLE_isMLE,
    ExtremeValueTheory.ae_tendsto_gpdScaleMLE', ExtremeValueTheory.tendstoInDistribution_gpdScaleMLE'}
  \leanok
  The supremum over rational admissible scales with nonnegative score is a measurable function of the sample that is
  almost surely the unique root and the unique likelihood maximizer; it is strongly consistent and
  $\sqrt n(\hat\sigma_n - \sigma_0) \Rightarrow N(0, \sigma_0^2(1+2\xi))$ for $\xi > -1/2$, $\xi \ne 0$, with no further hypotheses.
\end{theorem}
\begin{proof}
  \leanok
  The score is strictly decreasing in $\sigma$ and the admissible set is open.
\end{proof}

\begin{theorem}[Fisher information and joint asymptotic normality]
  \label{thm:gpd-joint-normality}
  \uses{thm:wald-ereal, thm:gpd-scale-mle-measurable}
  \lean{ExtremeValueTheory.inv_gpdFisherInfo, ExtremeValueTheory.tendstoInDistribution_gpdScoreVec, ExtremeValueTheory.tendstoInDistribution_gpdJointMLE,
    ExtremeValueTheory.tendstoInDistribution_gpdMLE, ExtremeValueTheory.score_eq_zero_of_isMaxOn}
  \leanok
  For $\xi > -1/2$, $\xi \ne 0$, the GPD score has mean zero and covariance
  $I(\xi,\sigma) = \frac{1}{(1+\xi)(1+2\xi)}\begin{pmatrix} 2 & 1/\sigma \ 1/\sigma & (1+\xi)/\sigma^2 \end{pmatrix}$, equal to minus the expected
  Hessian, with $I^{-1} = (1+\xi)\begin{pmatrix} 1+\xi & -\sigma \ -\sigma & 2\sigma^2 \end{pmatrix}$, and the normalized score is
  asymptotically $N(0, I)$. For $\xi_0 > 0$, measurable maximizers over a compact set with $\theta_0$ in its interior satisfy
  $\sqrt n(\hat\theta_n - \theta_0) \Rightarrow N(0, I(\theta_0)^{-1})$ [Smith 1985], [Coles, \S4.3]. The joint result for
  $-1/2 < \xi_0 < 0$ (moving support) is not formalized.
\end{theorem}
\begin{proof}
  \leanok
  An exact integral form of the mean-value expansion, a Lipschitz bound on the Hessian near $\theta_0$ (uses $\xi_0 > 0$), the strong
  law for the averaged observed information, Cram\'er--Wold for the score, and Slutsky.
\end{proof}

\begin{theorem}[Consistency of the GEV maximum likelihood estimator]
  \label{thm:gev-mle}
  \uses{thm:wald-ereal, thm:gpd-gev-density}
  \lean{ExtremeValueTheory.gevLogDensityE, ExtremeValueTheory.gevLogLikE, ExtremeValueTheory.gevDensity_eq_exp, ExtremeValueTheory.continuousAt_gevDensity,
    ExtremeValueTheory.continuousOn_gevLogDensityE, ExtremeValueTheory.gevLogDensityE_le, ExtremeValueTheory.measure_gevLocScaleCDF_eq_withDensity,
    ExtremeValueTheory.gev_params_eq_of_measure_eq, ExtremeValueTheory.integral_log_gevLocScaleDensity_lt, ExtremeValueTheory.exists_isMaxOn_gevLogLikE,
    ExtremeValueTheory.ae_tendsto_gevMLE}
  \leanok
  The GEV log-density ($-\infty$ off the support) is continuous in $(\xi, \mu, \sigma)$ on $\{\sigma > 0, \xi > -1\}$, including across
  $\xi = 0$; the three-parameter family is identifiable; and approximate maximizers of the likelihood over a compact subset
  of $\{\xi > -1, \sigma > 0\}$ are strongly consistent [Smith 1985], [B\"ucher and Segers 2017].
\end{theorem}
\begin{proof}
  \leanok
  Write the density as $\exp(-(1+\xi)s - e^{-s})$ with $s = z\log(1+\xi z)/(\xi z)$; the envelope $-\log\sigma + (1+\xi)^2$;
  identifiability from the derivative of $-\log(-\log F)$ near $\mu$; the extended-real Wald theorem.
\end{proof}

\begin{theorem}[GEV Fisher information for location and scale]
  \label{thm:gev-fisher}
  \uses{thm:gev-mle, thm:gev-moments}
  \lean{ExtremeValueTheory.gevScoreMu, ExtremeValueTheory.gevScoreSigma, ExtremeValueTheory.integral_gevPoly, ExtremeValueTheory.gevFisherInfoMuSigma,
    ExtremeValueTheory.integral_lin_gevScore_sq, ExtremeValueTheory.Gamma_one_add_sq_lt, ExtremeValueTheory.posDef_gevFisherInfoMuSigma,
    ExtremeValueTheory.tendstoInDistribution_gevScoreVec, ExtremeValueTheory.tendstoInDistribution_lin_gevScore}
  \leanok
  For known shape $\xi > -1/2$, $\xi \ne 0$, with $p = (1+\xi)^2\Gamma(1+2\xi)$: $I_{\mu\mu} = p/\sigma^2$,
  $I_{\mu\sigma} = (\Gamma(2+\xi) - p)/(\xi\sigma^2)$, $I_{\sigma\sigma} = (1 - 2\Gamma(2+\xi) + p)/(\xi^2\sigma^2)$ [Prescott and Walden 1980];
  this matrix is positive definite and the normalized $(\mu, \sigma)$ score is asymptotically $N(0, I)$. The shape row (which
  involves digamma values), the Gumbel case, and joint normality of the GEV MLE are not formalized.
\end{theorem}
\begin{proof}
  \leanok
  Moments $\E[T^k (T^\xi)^j] = \Gamma(k + j\xi + 1)$ for $T$ standard exponential; $\Gamma(1+\xi)^2 < \Gamma(1+2\xi)$; Cram\'er--Wold.
\end{proof}

\begin{theorem}[Joint normality of the GPD MLE for all regular shapes]
  \label{thm:gpd-joint-normality-all}
  \uses{thm:gpd-joint-normality}
  \lean{ExtremeValueTheory.gpdHess_quad_le, ExtremeValueTheory.tendstoInDistribution_gpdJointMLE_neg, ExtremeValueTheory.exists_measurable_isMaxOn,
    ExtremeValueTheory.tendstoInDistribution_gpdJointMLE_all, ExtremeValueTheory.tendstoInDistribution_gpdMLE_all, ExtremeValueTheory.score_eq_zero_of_isMaxOn',
    ExtremeValueTheory.exists_measurable_gpdMLE, ExtremeValueTheory.exists_gpdMLE_tendstoInDistribution}
  \leanok
  For $\xi_0 > -1/2$, $\xi_0 \ne 0$, a measurable maximum likelihood estimator over a compact set with $\theta_0$ in its interior
  exists and satisfies $\sqrt n(\hat\theta_n - \theta_0) \Rightarrow N(0, I(\theta_0)^{-1})$ [Smith 1985]. The case $\xi_0 = 0$ is not
  formalized.
\end{theorem}
\begin{proof}
  \leanok
  For $-1/2 < \xi_0 < 0$ the support moves with the parameter. A one-sided Hessian bound $t^\top H t \le C|t|^2$ (the terms
  that blow up at the endpoint have a favourable sign) gives eventual uniform positive definiteness of the averaged
  information, hence a $\sqrt n$ rate; with $\max_i 1/(\sigma_0 + \xi_0 X_i) = o_P(\sqrt n)$ all observations stay well inside the
  estimated support, and the argument for $\xi_0 > 0$ applies. Measurability: the lexicographically least maximizer over a
  compact set is measurable (Jennrich 1969).
\end{proof}

\section{Return levels and confidence intervals}

\begin{theorem}[The delta method]
  \label{thm:delta-method}
  \lean{MeasureTheory.TendstoInDistribution.delta_method, MeasureTheory.TendstoInDistribution.delta_method_gaussian,
    MeasureTheory.TendstoInDistribution.tendstoInMeasure_of_smul, ProbabilityTheory.map_inner_multivariateGaussian}
  \leanok
  If $r_n(\hat\theta_n - \theta) \Rightarrow Z$ with $r_n \to \infty$ and $g$ is differentiable at $\theta$, then
  $r_n(g(\hat\theta_n) - g(\theta)) \Rightarrow g'(\theta) Z$; for $Z \sim N(0, S)$ and a gradient $a$ the limit is $N(0, a^\top S a)$
  [van der Vaart, Theorem~3.1].
\end{theorem}
\begin{proof}
  \leanok
  Tightness of $r_n(\hat\theta_n - \theta)$, a first-order expansion, and Slutsky's lemma.
\end{proof}

\begin{theorem}[Asymptotic normality of return levels]
  \label{thm:return-level-normality}
  \uses{thm:delta-method}
  \lean{ExtremeValueTheory.gevReturnLevel, ExtremeValueTheory.gpdReturnLevel, ExtremeValueTheory.tendstoInDistribution_gpdReturnLevel, ExtremeValueTheory.tendstoInDistribution_gevReturnLevel,
    ExtremeValueTheory.tendstoInDistribution_gumbelReturnLevel, ExtremeValueTheory.gpdReturnLevelVar_pos}
  \leanok
  The GEV return level $z_m = \mu - \frac{\sigma}{\xi}(1 - y_m^{-\xi})$, $y_m = -\log(1 - 1/m)$, and the peaks-over-threshold
  return level $x_N = u + \frac{\sigma}{\xi}((N\zeta_u)^\xi - 1)$ are asymptotically normal with variance $\nabla^\top V \nabla$ [Coles,
  Sections~3.3.3 and~4.3.3]. For the GPD the result is unconditional, from the joint normality of the maximum likelihood
  estimator ($\xi > -1/2$, $\xi \ne 0$, $\zeta_u$ known); for the GEV it assumes normality of the parameter estimator.
\end{theorem}
\begin{proof}
  \leanok
  The delta method with the explicit gradients of the return level maps.
\end{proof}

\begin{theorem}[Wald confidence intervals]
  \label{thm:wald-intervals}
  \uses{thm:delta-method, thm:return-level-normality}
  \lean{MeasureTheory.TendstoInDistribution.tendsto_measure_mem_waldInterval,
    MeasureTheory.TendstoInDistribution.tendsto_measure_mem_waldInterval_gaussian, ProbabilityTheory.exists_gaussianReal_Icc_eq,
    ExtremeValueTheory.tendsto_measure_mem_hillInterval, ExtremeValueTheory.tendsto_measure_mem_gpdReturnLevelInterval, ExtremeValueTheory.tendsto_measure_mem_gevReturnLevelInterval}
  \leanok
  If $r_n(\hat\theta_n - \theta) \Rightarrow N(0, s^2)$ and $\hat s_n \to s > 0$ in probability, the interval
  $\hat\theta_n \pm z\hat s_n/r_n$ has asymptotic coverage $N(0,1)([-z, z])$, which is $1 - \alpha$ for the right $z$. This applies
  to the Hill estimator ($\hat\gamma \pm z\hat\gamma/\sqrt k$, without bias) and to peaks-over-threshold return levels with
  the plug-in standard error.
\end{theorem}
\begin{proof}
  \leanok
  Slutsky's lemma and the portmanteau theorem.
\end{proof}

\begin{theorem}[L-moments of the GPD and GEV]
  \label{thm:l-moments}
  \lean{ExtremeValueTheory.pwm, ExtremeValueTheory.pwmSurv, ExtremeValueTheory.lMoment1, ExtremeValueTheory.lMoment2, ExtremeValueTheory.lMoment3, ExtremeValueTheory.lSkewness, ExtremeValueTheory.pwmSurv_gpdMeasure,
    ExtremeValueTheory.lSkewness_gpdMeasure, ExtremeValueTheory.gpdPWMShape_lMoment, ExtremeValueTheory.gpdPWMScale_lMoment, ExtremeValueTheory.lSkewness_gevCDF_hosking}
  \leanok
  For the GPD with $\xi < 1$, $\alpha_r = \E[X(1-F(X))^r] = \sigma/((r+1)(r+1-\xi))$, $\lambda_1 = \sigma/(1-\xi)$,
  $\lambda_2 = \sigma/((1-\xi)(2-\xi))$ and $\tau_3 = (1+\xi)/(3-\xi)$, so $\xi = 2 - \lambda_1/\lambda_2$ and $\sigma = \lambda_1(\lambda_1/\lambda_2 - 1)$.
  For the GEV with $\xi < 1$, $\xi \ne 0$, $\lambda_2 = (2^\xi - 1)\Gamma(1-\xi)/\xi$ and $\tau_3 = (2 \cdot 3^\xi - 3 \cdot 2^\xi + 1)/(2^\xi - 1)$
  [Hosking, Wallis and Wood 1985; Hosking 1990, with $k = -\xi$].
\end{theorem}
\begin{proof}
  \leanok
  Direct integration against the GPD survival function and the GEV distribution function.
\end{proof}

\begin{theorem}[The mean excess function]
  \label{thm:mean-excess}
  \lean{ExtremeValueTheory.meanExcess_measure_eq, ExtremeValueTheory.meanExcess_gpdMeasure, ExtremeValueTheory.empMeanExcess, ExtremeValueTheory.ae_tendsto_empMeanExcess,

  \leanok

\end{proof}

\section{The full GEV likelihood}

\begin{theorem}[Fisher information of the three-parameter GEV]
  \label{thm:gev-fisher-3}
  \lean{ExtremeValueTheory.gevFisherInfo, ExtremeValueTheory.gevScoreXi, ExtremeValueTheory.hasDerivAt_log_gevLocScaleDensity_xi, ExtremeValueTheory.integral_gevScoreXi,
    ExtremeValueTheory.integral_gevScore_mul_gevScore, ExtremeValueTheory.gevFisherInfo_submatrix_castLE, ExtremeValueTheory.posDef_gevFisherInfo,
    ExtremeValueTheory.tendstoInDistribution_gevScoreVec3, ExtremeValueTheory.integral_gevHess_eq_neg_gevFisherInfo}
  \leanok
  For $\sigma > 0$, $\xi > -1/2$, $\xi \ne 0$, the scores of the GEV$(\mu, \sigma, \xi)$ log-density have mean zero, their covariance is the
  closed-form $3 \times 3$ Fisher information (in terms of $\Gamma$, the digamma function and Euler's constant), which is positive
  definite and equals minus the expected Hessian, and the normalized score sum is asymptotically $N(0, I)$
  [Prescott and Walden 1980; Smith 1985].
\end{theorem}
\begin{proof}
  \leanok
  Substitute $t = (1 + \xi z)^{-1/\xi}$ and evaluate moments of $t^a (\log t)^b e^{-t}$ through derivatives of $\Gamma$. Positive
  definiteness: $(t-1)\log t$ is not a combination of $1, t, t^\xi, t^{\xi+1}$ on $(0,\infty)$.
\end{proof}

\begin{theorem}[The Gumbel case \texorpdfstring{$\xi = 0$}{xi = 0}]
  \label{thm:gev-gumbel-limit}
  \uses{thm:gev-fisher-3}
  \lean{ExtremeValueTheory.tendsto_gevScoreMu_zero, ExtremeValueTheory.tendsto_gevScoreSigma_zero, ExtremeValueTheory.tendsto_gevScoreXi_zero,
    ExtremeValueTheory.continuousAt_log_gevLocScaleDensity_xi_zero, ExtremeValueTheory.hasDerivAt_log_gevLocScaleDensity_xi_zero,
    ExtremeValueTheory.tendsto_gpdScoreXi_zero, ExtremeValueTheory.hasDerivAt_log_gpdDensity_xi_zero, ExtremeValueTheory.gumbelFisherInfo, ExtremeValueTheory.posDef_gumbelFisherInfo,
    ExtremeValueTheory.tendstoInDistribution_gumbelScoreVec}
  \leanok
  The GEV and GPD log-densities and scores extend continuously to $\xi = 0$, where the extended scores are the true derivatives
  (the shape score of the Gumbel law is $-z + z^2(1 - e^{-z})/2$). The Gumbel Fisher information in $(\mu, \sigma)$ is
  $\sigma^{-2}\begin{pmatrix} 1 & -(1-\gamma) \ -(1-\gamma) & \pi^2/6 + (1-\gamma)^2 \end{pmatrix}$, positive definite.
\end{theorem}
\begin{proof}
  \leanok
  Second-order expansions of $\log(1 + \xi z)$ in $\xi$.
\end{proof}
